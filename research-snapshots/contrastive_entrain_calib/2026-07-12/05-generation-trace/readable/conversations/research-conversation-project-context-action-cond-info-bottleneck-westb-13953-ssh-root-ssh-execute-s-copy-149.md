@staticmethod
def diversity_regularizer(logits):
    """Penalize low variance in scorer outputs to prevent collapse to constant.
    
    Returns negative std (to be maximized, i.e., added as -lambda * this to total loss).

    p = torch.sigmoid(logits)           # [B, T]
    return -p.std(dim=-1).mean()        # scalar, negative so minimizing = maximizing std
