                _, protected_logits = self._adversary_model(
                    pred_logits, self.true_labels_ph)
                # Balance both protected classes in every minibatch.  This
                # prevents the adversary from learning the majority group
                # while giving the classifier a stronger, less biased signal.
                protected_rate = tf.reduce_mean(self.protected_attributes_ph)
                protected_weights = tf.where(
                    self.protected_attributes_ph > 0.5,
                    tf.ones_like(self.protected_attributes_ph) *
                    (0.5 / (protected_rate + 1e-6)),
                    tf.ones_like(self.protected_attributes_ph) *
                    (0.5 / (1.0 - protected_rate + 1e-6)))
                protected_bce = tf.nn.sigmoid_cross_entropy_with_logits(
                    labels=self.protected_attributes_ph, logits=protected_logits)
                protected_loss = tf.reduce_mean(protected_bce * protected_weights)

            global_step = tf.Variable(0, trainable=False)
            learning_rate = tf.train.exponential_decay(
                0.001, global_step, 1000, 0.96, staircase=True)
            classifier_opt = tf.train.AdamOptimizer(learning_rate)
            if self.debias:
                adversary_opt = tf.train.AdamOptimizer(learning_rate)
            classifier_vars = [
                v for v in tf.trainable_variables(scope=self.scope_name)
                if 'classifier_model' in v.name]
            classifier_gradients = classifier_opt.compute_gradients(
                pred_labels_loss, var_list=classifier_vars)
            if self.debias:
                adversary_vars = [
                    v for v in tf.trainable_variables(scope=self.scope_name)
                    if 'adversary_model' in v.name]
                # These gradients must be taken with respect to classifier
                # variables so they can be used for the debiasing update.
                classifier_adversary_gradients = adversary_opt.compute_gradients(
                    protected_loss, var_list=classifier_vars)
                adversary_gradients = adversary_opt.compute_gradients(
                    protected_loss, var_list=adversary_vars)

            normalize = lambda x: x / (tf.norm(x) + np.finfo(np.float32).tiny)
            classifier_grads = []
            for (grad, var), (agrad, _) in zip(
                    classifier_gradients,
                    classifier_adversary_gradients if self.debias else
                    [(None, var) for var in classifier_vars]):
                if grad is not None and self.debias and agrad is not None:
                    # Prevent the adversarial term from overwhelming the
                    # supervised classification gradient.
                    grad_norm = tf.norm(grad)
                    agrad_norm = tf.norm(agrad)
                    agrad = agrad * tf.minimum(
                        1.0, grad_norm / (agrad_norm + np.finfo(np.float32).tiny))
                    agrad_unit = normalize(agrad)
                    grad -= tf.reduce_sum(grad * agrad_unit) * agrad_unit
                    grad -= self.adversary_loss_weight * agrad
                classifier_grads.append((grad, var))
            classifier_minimizer = classifier_opt.apply_gradients(
                classifier_grads, global_step=global_step)
            if self.debias:
                with tf.control_dependencies([classifier_minimizer]):
                    adversary_minimizer = adversary_opt.apply_gradients(
                        adversary_gradients)

            self.sess.run(tf.global_variables_initializer())
            self.sess.run(tf.local_variables_initializer())
            protected_index = dataset.protected_attribute_names.index(
                self.protected_attribute_name)
            for epoch in range(self.num_epochs):
                shuffled_ids = np.random.choice(
                    num_train_samples, num_train_samples, replace=False)
                for i in range(num_train_samples // self.batch_size):
                    ids = shuffled_ids[i * self.batch_size:(i + 1) * self.batch_size]
                    batch_features = dataset.features[ids]
                    batch_labels = np.reshape(temp_labels[ids], [-1, 1])
                    batch_protected = np.reshape(
                        dataset.protected_attributes[ids, protected_index], [-1, 1])
                    feed = {self.features_ph: batch_features,
                            self.true_labels_ph: batch_labels,
                            self.protected_attributes_ph: batch_protected,
                            self.keep_prob: 0.8}
                    if self.debias:
                        self.sess.run([classifier_minimizer, adversary_minimizer],
                                      feed_dict=feed)
                    else:
                        self.sess.run(classifier_minimizer, feed_dict=feed)
            self._learn_group_thresholds(
                dataset.features, np.reshape(temp_labels, [-1, 1]),
                np.reshape(dataset.protected_attributes[:, protected_index], [-1, 1]))
        return self

    def predict(self, dataset):
        if self.seed is not None:
            np.random.seed(self.seed)
        num_test_samples = len(dataset.features)
        pred_labels = []
        protected_index = dataset.protected_attribute_names.index(
            self.protected_attribute_name)
        for start in range(0, num_test_samples, self.batch_size):
            end = min(start + self.batch_size, num_test_samples)
            ids = np.arange(start, end)
            feed = {self.features_ph: dataset.features[ids],
                    self.true_labels_ph: np.reshape(dataset.labels[ids], [-1, 1]),
                    self.protected_attributes_ph: np.reshape(
                        dataset.protected_attributes[ids, protected_index], [-1, 1]),
                    self.keep_prob: 1.0}
            pred_labels.extend(self.sess.run(self.pred_labels, feed_dict=feed)[:, 0])
        scores = np.asarray(pred_labels, dtype=np.float64)
        protected = dataset.protected_attributes[:, protected_index]
        uvalue = self.unprivileged_groups[0][self.protected_attribute_name]
        pvalue = self.privileged_groups[0][self.protected_attribute_name]
        thresholds = np.full(num_test_samples, 0.5, dtype=np.float64)
        thresholds[np.isclose(protected, uvalue)] = self.unprivileged_threshold
        thresholds[np.isclose(protected, pvalue)] = self.privileged_threshold
        dataset_new = dataset.copy(deepcopy=True)
        dataset_new.scores = scores.reshape(-1, 1)
        dataset_new.labels = (scores > thresholds).astype(np.float64).reshape(-1, 1)
        dataset_new.labels[dataset_new.labels == 1.0] = dataset.favorable_label
        dataset_new.labels[dataset_new.labels == 0.0] = dataset.unfavorable_label
        return dataset_new
