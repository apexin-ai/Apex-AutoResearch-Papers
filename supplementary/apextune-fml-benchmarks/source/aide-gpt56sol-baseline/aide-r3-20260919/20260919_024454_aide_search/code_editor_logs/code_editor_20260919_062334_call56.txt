            np.random.seed(self.seed)
        ii32 = np.iinfo(np.int32)
        self.seed1, self.seed2, self.seed3, self.seed4 = np.random.randint(
            ii32.min, ii32.max, size=4)
        labels = dataset.labels.copy()
        labels[(dataset.labels == dataset.favorable_label).ravel(), 0] = 1.0
        labels[(dataset.labels == dataset.unfavorable_label).ravel(), 0] = 0.0
        with tf.variable_scope(self.scope_name):
            n, self.features_dim = np.shape(dataset.features)
            self.features_ph = tf.placeholder(tf.float32, [None, self.features_dim])
            self.protected_attributes_ph = tf.placeholder(tf.float32, [None, 1])
            self.true_labels_ph = tf.placeholder(tf.float32, [None, 1])
            self.keep_prob = tf.placeholder(tf.float32)
            self.pred_labels, logits = self._classifier_model(
                self.features_ph, self.features_dim, self.keep_prob)
            classification_loss = tf.reduce_mean(
                tf.nn.sigmoid_cross_entropy_with_logits(
                    labels=self.true_labels_ph, logits=logits))

            if self.debias:
                _, adversary_logits = self._adversary_model(
                    logits, self.true_labels_ph)
                adversary_loss = tf.reduce_mean(
                    tf.nn.sigmoid_cross_entropy_with_logits(
                        labels=self.protected_attributes_ph,
                        logits=adversary_logits))
                odds_penalty = self._soft_equalized_odds_penalty(self.pred_labels)

                # The classifier maximizes adversary error while minimizing
                # classification and equalized-odds losses.
                classifier_loss = (
                    classification_loss
                    - self.adversary_loss_weight * adversary_loss
                    + self.adversary_loss_weight * odds_penalty)
            else:
                classifier_loss = classification_loss

            step = tf.Variable(0, trainable=False)
            lr = tf.train.exponential_decay(
                0.001, step, 1000, 0.96, staircase=True)
            copt = tf.train.AdamOptimizer(lr)
            classifier_vars = [
                v for v in tf.trainable_variables(scope=self.scope_name)
                if 'classifier_model' in v.name
            ]

            if self.debias:
                aopt = tf.train.AdamOptimizer(lr)
                adversary_vars = [
                    v for v in tf.trainable_variables(scope=self.scope_name)
                    if 'adversary_model' in v.name
                ]

            cmin = copt.minimize(
                classifier_loss, var_list=classifier_vars, global_step=step)

            if self.debias:
                with tf.control_dependencies([cmin]):
                    amin = aopt.minimize(
                        adversary_loss, var_list=adversary_vars)

            self.sess.run(tf.global_variables_initializer())
            self.sess.run(tf.local_variables_initializer())
            protected_index = dataset.protected_attribute_names.index(
                self.protected_attribute_name)
            for _ in range(self.num_epochs):
                ids = np.random.permutation(n)
                for start in range(0, n - self.batch_size + 1, self.batch_size):
                    ids_batch = ids[start:start + self.batch_size]
                    bf = dataset.features[ids_batch]
                    bl = labels[ids_batch].reshape(-1, 1)
                    bp = dataset.protected_attributes[
                        ids_batch, protected_index].reshape(-1, 1)
                    feed = {
                        self.features_ph: bf,
                        self.true_labels_ph: bl,
                        self.protected_attributes_ph: bp,
                        self.keep_prob: 0.8
                    }
                    if self.debias:
                        self.sess.run([cmin, amin], feed_dict=feed)
                    else:
                        self.sess.run(cmin, feed_dict=feed)

            training_protected = dataset.protected_attributes[
                :, protected_index].reshape(-1, 1)
            self._learn_group_thresholds(
                dataset.features, labels.reshape(-1, 1),
                training_protected)
        return self

    def predict(self, dataset):
        if self.seed is not None:
            np.random.seed(self.seed)
        n = len(dataset.features)
        scores = []
        protected_index = dataset.protected_attribute_names.index(
            self.protected_attribute_name)
        for start in range(0, n, self.batch_size):
            end = min(start + self.batch_size, n)
            feed = {
                self.features_ph: dataset.features[start:end],
                self.true_labels_ph: np.zeros(
                    (end - start, 1), dtype=np.float32),
                self.protected_attributes_ph: dataset.protected_attributes[
                    start:end, protected_index].reshape(-1, 1),
                self.keep_prob: 1.0
            }
            scores.extend(
                self.sess.run(self.pred_labels, feed_dict=feed)[:, 0])

        scores = np.asarray(scores, dtype=np.float64)
        protected = dataset.protected_attributes[:, protected_index]
        uv = self.unprivileged_groups[0][self.protected_attribute_name]
        pv = self.privileged_groups[0][self.protected_attribute_name]
        thresholds = np.full(n, 0.5, dtype=np.float64)
        thresholds[np.isclose(protected, uv)] = self.unprivileged_threshold
        thresholds[np.isclose(protected, pv)] = self.privileged_threshold

        result = dataset.copy(deepcopy=True)
        result.scores = scores.reshape(-1, 1)
        result.labels = (scores > thresholds).astype(np.float64).reshape(-1, 1)
        result.labels[result.labels == 1.0] = dataset.favorable_label
        result.labels[result.labels == 0.0] = dataset.unfavorable_label
        return result
