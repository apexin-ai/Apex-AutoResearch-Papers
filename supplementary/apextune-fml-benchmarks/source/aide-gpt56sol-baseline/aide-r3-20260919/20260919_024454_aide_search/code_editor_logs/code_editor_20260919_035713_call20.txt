class AdversarialDebiasing(Transformer):
    """Adversarial debiasing with a conservative differentiable
    equalized-odds penalty.

    The direct penalty is guarded against minibatches which do not contain
    every protected-label combination. This avoids very large gradients caused
    by dividing by an effectively zero minibatch count.

                 sess, seed=None, adversary_loss_weight=0.01, num_epochs=50,
                 batch_size=128, classifier_num_hidden_units=200,

            privileged_groups=privileged_groups)

        self.num_epochs = num_epochs
        self.batch_size = batch_size
        self.classifier_num_hidden_units = classifier_num_hidden_units
        self.debias = debias

        self.features_ph = None
        self.protected_attributes_ph = None
        self.true_labels_ph = None
        self.pred_labels = None
        self.unprivileged_threshold = 0.5
        self.privileged_threshold = 0.5

            w1 = tf.get_variable(
                'W1', [features_dim, self.classifier_num_hidden_units],
                initializer=tf.initializers.glorot_uniform(seed=self.seed1))
            b1 = tf.Variable(tf.zeros([self.classifier_num_hidden_units]), name='b1')

                               keep_prob=keep_prob, seed=self.seed2)
            w2 = tf.get_variable(
                'W2', [self.classifier_num_hidden_units, 1],
                initializer=tf.initializers.glorot_uniform(seed=self.seed3))
            b2 = tf.Variable(tf.zeros([1]), name='b2')
            logit = tf.matmul(h1, w2) + b2
            return tf.sigmoid(logit), logit

    def _adversary_model(self, pred_logits, true_labels):
        with tf.variable_scope("adversary_model"):
            c = tf.get_variable('c', initializer=tf.constant(1.0))
            s = tf.sigmoid((1 + tf.abs(c)) * pred_logits)
            w2 = tf.get_variable(
                'W2', [3, 1],
                initializer=tf.initializers.glorot_uniform(seed=self.seed4))
            b2 = tf.Variable(tf.zeros([1]), name='b2')
            x = tf.concat([s, s * true_labels, s * (1.0 - true_labels)], 1)
            logit = tf.matmul(x, w2) + b2
            return tf.sigmoid(logit), logit

    def _equalized_odds_penalty(self, scores):
        """Soft squared TPR/FPR gap with safe minibatch normalization."""
        protected = self.protected_attributes_ph
        labels = self.true_labels_ph
        unpriv = tf.cast(tf.equal(
            protected, float(self.unprivileged_groups[0][
                self.protected_attribute_name])), tf.float32)
        priv = tf.cast(tf.equal(
            protected, float(self.privileged_groups[0][
                self.protected_attribute_name])), tf.float32)
        eps = tf.constant(1e-6, tf.float32)

        def conditional_mean(values, mask):
            return tf.reduce_sum(values * mask) / tf.maximum(
                tf.reduce_sum(mask), tf.constant(1.0, tf.float32))

        unpriv_pos = unpriv * labels
        priv_pos = priv * labels
        unpriv_neg = unpriv * (1.0 - labels)
        priv_neg = priv * (1.0 - labels)

                   conditional_mean(scores, priv_pos))
        fpr_gap = (conditional_mean(scores, unpriv_neg) -
                   conditional_mean(scores, priv_neg))

            tf.reduce_sum(unpriv_pos) > eps,
            tf.reduce_sum(priv_pos) > eps), tf.float32)
        fpr_valid = tf.cast(tf.logical_and(
            tf.reduce_sum(unpriv_neg) > eps,
            tf.reduce_sum(priv_neg) > eps), tf.float32)
        return 0.5 * (tpr_valid * tf.square(tpr_gap) +
                      fpr_valid * tf.square(fpr_gap))

        # can sharply reduce held-out accuracy. Keep the standard classifier

            self.sess.run(tf.local_variables_initializer())

            for _ in range(self.num_epochs):

                for i in range(n // self.batch_size):

                dataset.features, temp_labels.reshape(-1, 1),
                dataset.protected_attributes[:, pidx].reshape(-1, 1))
        return self

        uv = self.unprivileged_groups[0][self.protected_attribute_name]
        pv = self.privileged_groups[0][self.protected_attribute_name]
        pa = dataset.protected_attributes[:, pidx]
        thresholds = np.full(n, .5)
        thresholds[np.isclose(pa, uv)] = self.unprivileged_threshold
        thresholds[np.isclose(pa, pv)] = self.privileged_threshold
        result = dataset.copy(deepcopy=True)
        result.scores = scores.reshape(-1, 1)
        result.labels = (scores > thresholds).astype(np.float64).reshape(-1, 1)
        result.labels[result.labels == 1.] = dataset.favorable_label
        result.labels[result.labels == 0.] = dataset.unfavorable_label
        return result

                        # Apply the adversary signal directly and scale it by
                        # its configured weight, preserving the classifier
                        # gradient magnitude and direction otherwise.
                        grad -= self.adversary_loss_weight * ag
                cgrads.append((grad, var))
            cmin = copt.apply_gradients(cgrads, global_step=step)
            if self.debias:
                with tf.control_dependencies([cmin]):
                    amin = aopt.minimize(adversary_loss, var_list=avars)
            self.sess.run(tf.global_variables_initializer())
            self.sess.run(tf.local_variables_initializer())
            pidx = dataset.protected_attribute_names.index(self.protected_attribute_name)
            for _ in range(self.num_epochs):
                ids = np.random.choice(n, n, replace=False)
                for i in range(n // self.batch_size):
                    bid = ids[i * self.batch_size:(i + 1) * self.batch_size]
                    bl = temp_labels[bid].reshape(-1, 1)
                    bp = dataset.protected_attributes[bid, pidx].reshape(-1, 1)
                    feed = {self.features_ph: dataset.features[bid],
                            self.true_labels_ph: bl,
                            self.protected_attributes_ph: bp,
                            self.keep_prob: .8}
                    if self.debias:
                        self.sess.run([cmin, amin], feed_dict=feed)
                    else:
                        self.sess.run(cmin, feed_dict=feed)
            self._learn_group_thresholds(
                dataset.features, temp_labels.reshape(-1, 1),
                dataset.protected_attributes[:, pidx].reshape(-1, 1))
        return self

    def predict(self, dataset):
        if self.seed is not None:
            np.random.seed(self.seed)
        n = len(dataset.features)
        pidx = dataset.protected_attribute_names.index(self.protected_attribute_name)
        scores = []
        for start in range(0, n, self.batch_size):
            end = min(start + self.batch_size, n)
            ids = np.arange(start, end)
            feed = {self.features_ph: dataset.features[ids],
                    self.true_labels_ph: dataset.labels[ids].reshape(-1, 1),
                    self.protected_attributes_ph:
                        dataset.protected_attributes[ids, pidx].reshape(-1, 1),
                    self.keep_prob: 1.0}
            scores.extend(self.sess.run(self.pred_labels, feed_dict=feed)[:, 0])
        scores = np.asarray(scores, dtype=np.float64)
        uv = self.unprivileged_groups[0][self.protected_attribute_name]
        pv = self.privileged_groups[0][self.protected_attribute_name]
        pa = dataset.protected_attributes[:, pidx]
        thresholds = np.full(n, .5)
        thresholds[np.isclose(pa, uv)] = self.unprivileged_threshold
        thresholds[np.isclose(pa, pv)] = self.privileged_threshold
        result = dataset.copy(deepcopy=True)
        result.scores = scores.reshape(-1, 1)
        result.labels = (scores > thresholds).astype(np.float64).reshape(-1, 1)
        result.labels[result.labels == 1.] = dataset.favorable_label
        result.labels[result.labels == 0.] = dataset.unfavorable_label
        return result
