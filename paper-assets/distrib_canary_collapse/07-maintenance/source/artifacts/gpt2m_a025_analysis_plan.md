## Step 2: Lead-Lag Analysis (5-method)

Adapted from `leadlag_gpt2m_s42.py`. Creates `results/gen_wiki_gpt2m_a025_s42/analysis/leadlag_analysis.json`.

```bash
cd /root/autodl-tmp/distrib_canary_collapse
source /root/miniconda3/etc/profile.d/conda.sh && conda activate base
python3 << 'PYEOF'
import json, sys, os
import numpy as np

sys.path.insert(0, "/root/autodl-tmp/distrib_canary_collapse")
from permutation_granger import permutation_granger_test

DATA_PATH = "/root/autodl-tmp/distrib_canary_collapse/results/gen_wiki_gpt2m_a025_s42/all_metrics.json"
N_PERM = 5000
ALPHA = 0.05
SEED = 42

CANARIES = ["token_entropy", "ece"]
DOWNSTREAMS = ["distinct_1", "distinct_2", "distinct_3"]

WINDOWS = {
    "full_gen0_10": (0, 11),
    "sub_gen0_5": (0, 6),
}

def load_series(path):
    with open(path) as f:
        data = json.load(f)
    data.sort(key=lambda x: x["generation"])
    out = {}
    keys = ["token_entropy", "ece", "perplexity", "distinct_1", "distinct_2", "distinct_3"]
    for k in keys:
        out[k] = np.array([d[k] for d in data])
    return out

def cross_correlation(x, y, max_lag=5):
    n = len(x)
    results = []
    for lag in range(0, min(max_lag + 1, n)):
        if lag == 0:
            r = np.corrcoef(x, y)[0, 1]
        else:
            r = np.corrcoef(x[:-lag], y[lag:])[0, 1] if n - lag >= 3 else np.nan
        results.append({"lag": lag, "r": float(r) if not np.isnan(r) else None})
    best = max(results, key=lambda d: abs(d["r"]) if d["r"] is not None else -1)
    return results, best

def bh_fdr(p_values, alpha=0.05):
    pv = np.array(p_values, dtype=float)
    n = len(pv)
    valid = ~np.isnan(pv)
    adjusted = np.full(n, np.nan)
    if valid.sum() == 0:
        return adjusted.tolist(), [False] * n
    idx = np.where(valid)[0]
    p_valid = pv[idx]
    sorted_idx = np.argsort(p_valid)
    ranks = np.empty_like(sorted_idx)
    ranks[sorted_idx] = np.arange(1, len(p_valid) + 1)
    m = len(p_valid)
    adj = p_valid * m / ranks
    adj_sorted = adj[sorted_idx]
    for i in range(m - 2, -1, -1):
        adj_sorted[i + 1] = min(adj_sorted[i + 1], 1.0)
        adj_sorted[i] = min(adj_sorted[i], adj_sorted[i + 1])
    adj_sorted[m - 1] = min(adj_sorted[m - 1], 1.0)
    adj_final = np.empty(m)
    adj_final[sorted_idx] = adj_sorted
    for i, ix in enumerate(idx):
        adjusted[ix] = adj_final[i]
    significant = [(adjusted[i] < alpha if not np.isnan(adjusted[i]) else False) for i in range(n)]
    return adjusted.tolist(), significant

def run_analysis():
    series = load_series(DATA_PATH)
    results = {}
    for wname, (start, end) in WINDOWS.items():
        window_results = {"forward": [], "reverse": []}
        raw_p_forward, raw_p_reverse = [], []
        for canary in CANARIES:
            for downstream in DOWNSTREAMS:
                x_full = series[canary][start:end]
                y_full = series[downstream][start:end]
                max_xcorr_lag = min(5, len(x_full) - 3)
                # Forward
                xcorr_fwd, best_fwd = cross_correlation(x_full, y_full, max_lag=max_xcorr_lag)
                granger_fwd = permutation_granger_test(x_full, y_full, max_lag=1, n_perm=N_PERM, seed=SEED, difference=True)
                fwd_entry = {
                    "canary": canary, "downstream": downstream, "direction": "forward",
                    "xcorr": xcorr_fwd, "xcorr_best_r": best_fwd["r"], "xcorr_best_lag": best_fwd["lag"],
                    "granger_f": granger_fwd.get("f_stat"), "granger_p_raw": granger_fwd.get("p_value"),
                    "granger_f2": granger_fwd.get("effect_size_f2"), "df_resid": granger_fwd.get("df_resid"),
                }
                window_results["forward"].append(fwd_entry)
                raw_p_forward.append(granger_fwd.get("p_value") if granger_fwd.get("p_value") is not None else np.nan)
                # Reverse
                xcorr_rev, best_rev = cross_correlation(y_full, x_full, max_lag=max_xcorr_lag)
                granger_rev = permutation_granger_test(y_full, x_full, max_lag=1, n_perm=N_PERM, seed=SEED, difference=True)
                rev_entry = {
                    "canary": downstream, "downstream": canary, "direction": "reverse",
                    "xcorr": xcorr_rev, "xcorr_best_r": best_rev["r"], "xcorr_best_lag": best_rev["lag"],
                    "granger_f": granger_rev.get("f_stat"), "granger_p_raw": granger_rev.get("p_value"),
                    "granger_f2": granger_rev.get("effect_size_f2"), "df_resid": granger_rev.get("df_resid"),
                }
                window_results["reverse"].append(rev_entry)
                raw_p_reverse.append(granger_rev.get("p_value") if granger_rev.get("p_value") is not None else np.nan)
        # BH FDR correction
        fdr_adj_fwd, sig_fwd = bh_fdr(raw_p_forward, ALPHA)
        fdr_adj_rev, sig_rev = bh_fdr(raw_p_reverse, ALPHA)
        for i, entry in enumerate(window_results["forward"]):
            entry["granger_fdr_p"] = fdr_adj_fwd[i] if not np.isnan(fdr_adj_fwd[i]) else None
            entry["significant_fdr"] = sig_fwd[i]
        for i, entry in enumerate(window_results["reverse"]):
            entry["granger_fdr_p"] = fdr_adj_rev[i] if not np.isnan(fdr_adj_rev[i]) else None
            entry["significant_fdr"] = sig_rev[i]
        results[wname] = window_results
    # Print
    for wname in WINDOWS:
        T = WINDOWS[wname][1] - WINDOWS[wname][0]
        print(f"\n{'='*80}")
        print(f"WINDOW: {wname} (T={T})")
        print(f"{'='*80}")
        for direction in ["forward", "reverse"]:
            label = "FORWARD (canary -> diversity)" if direction == "forward" else "REVERSE (diversity -> canary)"
            print(f"\n  {label}")
            print(f"  {'Pair':<30} {'xcorr_r':>8} {'lag':>4} {'F':>8} {'p_raw':>8} {'p_fdr':>8} {'sig':>4} {'f2':>8} {'df_res':>6}")
            print(f"  {'-'*90}")
            for entry in results[wname][direction]:
                pair = f"{entry['canary']}->{entry['downstream']}"
                xcr = f"{entry['xcorr_best_r']:.4f}" if entry['xcorr_best_r'] is not None else "N/A"
                lag = str(entry['xcorr_best_lag'])
                F = f"{entry['granger_f']:.4f}" if entry['granger_f'] is not None else "N/A"
                pr = f"{entry['granger_p_raw']:.4f}" if entry['granger_p_raw'] is not None else "N/A"
                pf = f"{entry['granger_fdr_p']:.4f}" if entry['granger_fdr_p'] is not None else "N/A"
                sig = "*" if entry.get('significant_fdr') else ""
                f2 = f"{entry['granger_f2']:.4f}" if entry.get('granger_f2') is not None else "N/A"
                df = str(entry.get('df_resid', 'N/A'))
                print(f"  {pair:<30} {xcr:>8} {lag:>4} {F:>8} {pr:>8} {pf:>8} {sig:>4} {f2:>8} {df:>6}")
    # Save
    out_dir = "/root/autodl-tmp/distrib_canary_collapse/results/gen_wiki_gpt2m_a025_s42/analysis"
    os.makedirs(out_dir, exist_ok=True)
    out_path = os.path.join(out_dir, "leadlag_analysis.json")
    def clean(obj):
        if isinstance(obj, (np.floating, np.float64)): return float(obj)
        if isinstance(obj, (np.integer,)): return int(obj)
        if isinstance(obj, np.bool_): return bool(obj)
        if isinstance(obj, dict): return {k: clean(v) for k, v in obj.items()}
        if isinstance(obj, list): return [clean(v) for v in obj]
        return obj
    with open(out_path, "w") as f:
        json.dump(clean(results), f, indent=2)
    print(f"\nSaved: {out_path}")

if __name__ == "__main__":
    run_analysis()
PYEOF
```

| Null | 0.00 | <10% | <100% | <25% |

---

## Success / Negative Criteria

### SUCCESS (supports Claim 2: regime map generalizes cross-architecture)

All of:
1. **Drift in mild range**: entropy drift <20%, ECE drift <200%, PPL drift <50%
2. **Pattern consistent with Pythia α=0.25**: same direction of drift for all metrics, same order of magnitude
3. **Lead-lag structure proportional**: if signal detectable, entropy/ECE lead diversity (forward > reverse)
4. **Regime classification = MILD**: clearly distinct from GPT-2m α=1.0 (severe)

### NEGATIVE (challenges cross-architecture generalization)

Any of:
1. **Drift in wrong regime**: entropy >30% or ECE >500% (moderate/severe territory despite α=0.25)
2. **Opposite drift direction**: e.g. entropy increases while Pythia α=0.25 shows decrease
3. **No diversity collapse**: distinct metrics stay flat (<5% drift) while Pythia shows -40~60%
4. **Reversed causality**: diversity→canary stronger than canary→diversity

### INCONCLUSIVE

- Drift directions match but magnitudes differ >3x from Pythia α=0.25
- Lead-lag signal too weak to detect (all p>0.05) — expected given T=11 and mild contamination, not a failure per se
