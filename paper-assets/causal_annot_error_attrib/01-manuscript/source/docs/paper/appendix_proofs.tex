\section{Proofs of Proposition~\ref{prop:signflip} and Theorem~\ref{thm:monotone}}
\label{app:proof}

We provide algebraic proofs of sign preservation for the confounding and collider topologies (Proposition~\ref{prop:signflip}(a)--(b)), an explicit counterexample for exposure (Proposition~\ref{prop:signflip}(c)), numerical verification for mediation, M-bias, and front-door (Numerical Verification~1), and the monotonicity proof for confounding (Theorem~\ref{thm:monotone}).
Throughout, $A\in\{0,1,\dots,K{-}1\}$ with prior $p_k=P(A{=}k)$, and $C$ is a $K{\times}K$ column-stochastic confusion matrix with $C_{jk}=P(A^*{=}j\mid A{=}k)$.
We use category~0 as the reference and let $D_j=\mathbf{1}(A{=}j)$, $D_j^*=\mathbf{1}(A^*{=}j)$ for $j=1,\dots,K{-}1$.

\subsection{Notation and Key Identities}
\label{app:notation}

Under non-differential misclassification, the misclassified moments factor linearly through $C$.
Define the population vectors:
\begin{align}
  q_j &= P(A^*{=}j) = \textstyle\sum_k C_{jk}\,p_k, \label{eq:q}\\
  \mathrm{ct}_j &= \mathbb{E}[T\cdot D_j^*] = \textstyle\sum_k C_{jk}\,\mathbb{E}[T\cdot D_k], \label{eq:ct}\\
  \mathrm{cy}_j &= \mathbb{E}[Y\cdot D_j^*] = \textstyle\sum_k C_{jk}\,\mathbb{E}[Y\cdot D_k]. \label{eq:cy}
\end{align}
Write $q = Cp$, $\mathrm{ct} = C\,\mathbb{E}[T\cdot D]$, and $\mathrm{cy} = C\,\mathbb{E}[Y\cdot D]$, where $D=(D_0,\dots,D_{K-1})^\top$.

\subsection{Frisch--Waugh--Lovell Decomposition}
\label{app:fwl}

For topologies where $T$ is directly observed (confounding, mediation, collider, M-bias, front-door), the estimator regresses $Y$ on $Z^*=(1,T,D_1^*,\dots,D_{K-1}^*)$.
By the Frisch--Waugh--Lovell (FWL) theorem, the coefficient on $T$ satisfies:
\begin{equation}
  \operatorname{plim}\hat{\tau}(C)
  = \frac{\mathbb{E}[\tilde{T}\,\tilde{Y}]}{\mathbb{E}[\tilde{T}^2]},
  \label{eq:fwl}
\end{equation}
where $\tilde{T}=T-\mathbb{E}[T\mid A^*]$ and $\tilde{Y}=Y-\mathbb{E}[Y\mid A^*]$ are residuals from projecting onto $(1,D_1^*,\dots,D_{K-1}^*)$.

\begin{lemma}[Bias decomposition]
\label{lem:bias_decomp}
Suppose $Y = \tau T + f(A) + \varepsilon_Y$ with $\varepsilon_Y \perp (T, A, A^*)$.
Then
\begin{equation}
  \operatorname{plim}\hat\tau(C) = \tau + \frac{R(C)}{D(C)},
  \label{eq:bias_decomp}
\end{equation}
where
\begin{align}
  R(C) &= \mathbb{E}\bigl[\operatorname{Cov}(T,\,f(A)\mid A^*)\bigr], \label{eq:R}\\
  D(C) &= \mathbb{E}\bigl[\operatorname{Var}(T\mid A^*)\bigr] > 0. \label{eq:D}
\end{align}
\end{lemma}

\begin{proof}
Expand $\tilde{Y}$ using $Y=\tau T + f(A) + \varepsilon_Y$:
\[
  \tilde{Y} = \tau\,\tilde{T} + \bigl(f(A) - \mathbb{E}[f(A)\mid A^*]\bigr) + \varepsilon_Y.
\]
Since $\tilde{T}$ is measurable w.r.t.\ $(T,A^*)$ and $\varepsilon_Y\perp(T,A,A^*)$:
\[
  \mathbb{E}[\tilde{T}\,\tilde{Y}]
  = \tau\,\mathbb{E}[\tilde{T}^2]
  + \mathbb{E}\bigl[\tilde{T}\,\bigl(f(A)-\mathbb{E}[f(A)\mid A^*]\bigr)\bigr].
\]
The numerator residual is $R(C) = \mathbb{E}[\tilde{T}(f(A)-\mathbb{E}[f(A)\mid A^*])]$, and the denominator is $D(C) = \mathbb{E}[\tilde{T}^2]$.

By the law of total covariance:
\begin{align*}
  R(C) &= \sum_{j=0}^{K-1} q_j\,\operatorname{Cov}(T,\,f(A)\mid A^*{=}j)\\
  &= \mathbb{E}[\operatorname{Cov}(T,f(A)\mid A^*)].
\end{align*}
For $D(C)>0$: decompose $T = g(A) + \varepsilon_T$ where $g(A)=\mathbb{E}[T\mid A]$ (for confounding) or simply note that $T$ has exogenous variation independent of $A^*$ (for mediation).
In both cases, $\tilde{T}=\bigl(g(A)-\mathbb{E}[g(A)\mid A^*]\bigr)+\varepsilon_T'$, where $\varepsilon_T'$ is the component of $T$ independent of $(A,A^*)$.
Hence $D(C)\geq\operatorname{Var}(\varepsilon_T')>0$.
\end{proof}

\subsection{Confounding Topology ($A$ = Confounder)}
\label{app:confounding}

\paragraph{Data-generating process.}
$A\to T$, $A\to Y$, $T\to Y$.
Using dummy coding with category 0 as reference:
\begin{align*}
  T &= \textstyle\sum_{j=1}^{K-1}\alpha_j D_j + \varepsilon_T,\quad
  \varepsilon_T\sim\mathcal{N}(0,\sigma_T^2),\\
  Y &= \tau\, T + \textstyle\sum_{j=1}^{K-1}\beta_j^{(A)} D_j + \varepsilon_Y,\\
  &\quad \varepsilon_Y\sim\mathcal{N}(0,1),
\end{align*}
with $\varepsilon_T\perp A$ and $\varepsilon_Y\perp(T,A)$.
Define $g(k)=\sum_{j}\alpha_j\mathbf{1}(k{=}j)$ and
$f(k)=\sum_{j}\beta_j^{(A)}\mathbf{1}(k{=}j)$, so $g(0)=f(0)=0$.

\paragraph{Closed-form plim.}
Using the FWL decomposition, the coefficient on $T$ is:
\begin{equation}
  \operatorname{plim}\hat\tau(C)
  = \frac{\mu_{TY} - \sum_{j=0}^{K-1}\frac{\mathrm{ct}_j\,\mathrm{cy}_j}{q_j}}
  {s_T - \sum_{j=0}^{K-1}\frac{\mathrm{ct}_j^2}{q_j}},
  \label{eq:plim_conf}
\end{equation}
where $s_T=\mathbb{E}[T^2]$, $\mu_{TY}=\mathbb{E}[TY]$, and $\mathrm{ct}_j$, $\mathrm{cy}_j$, $q_j$ are defined in~\eqref{eq:q}--\eqref{eq:cy}.

\begin{proof}[Derivation]
By FWL, $\operatorname{plim}\hat\tau = \mathbb{E}[\tilde{T}Y]/\mathbb{E}[\tilde{T}^2]$ where $\tilde{T}=T-\mathbb{E}[T\mid A^*]$.
Since $\mathbb{E}[T\mid A^*{=}j]=\mathrm{ct}_j/q_j$:
\begin{align*}
  \mathbb{E}[\tilde{T}^2]
  &= \mathbb{E}[T^2] - \textstyle\sum_j q_j\bigl(\mathrm{ct}_j/q_j\bigr)^2\\
  &= s_T - \textstyle\sum_j \mathrm{ct}_j^2/q_j,\\
  \mathbb{E}[\tilde{T}\,Y]
  &= \mathbb{E}[TY] - \textstyle\sum_j \bigl(\mathrm{ct}_j/q_j\bigr)\,\mathbb{E}[Y\cdot D_j^*]\\
  &= \mu_{TY} - \textstyle\sum_j \mathrm{ct}_j\,\mathrm{cy}_j/q_j.
\end{align*}
\end{proof}

\paragraph{Verification at $C=I$.}
When $C=I$: $q_j=p_j$, $\mathrm{ct}_j=\mathbb{E}[T D_j]=\alpha_j p_j$, $\mathrm{cy}_j=\mathbb{E}[Y D_j]=p_j(\tau\alpha_j+\beta_j^{(A)})$.
Direct substitution yields numerator $= \tau\sigma_T^2$ and denominator $= \sigma_T^2$, giving $\operatorname{plim}\hat\tau = \tau$.

\paragraph{Bias structure.}
By Lemma~\ref{lem:bias_decomp}:
\begin{align}
  \operatorname{plim}\hat\tau(C) &= \tau + B(C),\notag\\
  B(C) &= \frac{\mathbb{E}[\operatorname{Cov}(g(A),\,f(A)\mid A^*)]}{D(C)}.
  \label{eq:bias_conf}
\end{align}
By the law of total covariance:
\begin{multline}
  \mathbb{E}[\operatorname{Cov}(g,f\mid A^*)]
  = \operatorname{Cov}(g(A),f(A))\\
  - \operatorname{Cov}\bigl(\mathbb{E}[g(A)\mid A^*],\;\mathbb{E}[f(A)\mid A^*]\bigr).
  \label{eq:lotc}
\end{multline}

At $C=I$: $\mathbb{E}[g\mid A^*]=g(A)$, so the between-group covariance equals $\operatorname{Cov}(g,f)$ and $B(I)=0$.

At $C=\frac{1}{K}\mathbf{1}\mathbf{1}^\top$ (uniform, $A^*\perp A$): $\mathbb{E}[g\mid A^*]$ is constant, so the between-group covariance is zero and
\begin{equation}
  B_{\mathrm{OV}} \;\equiv\; B\bigl(\tfrac{1}{K}\mathbf{1}\mathbf{1}^\top\bigr)
  = \frac{\operatorname{Cov}(g(A),\,f(A))}{\operatorname{Var}(T)},
  \label{eq:bov}
\end{equation}
which equals the classical omitted-variable bias (no control for $A$).

\paragraph{Sign preservation condition.}
Since $D(C)>0$, sign$(\operatorname{plim}\hat\tau) = $ sign$(\tau\cdot D(C) + R(C))$.
For sign preservation, it suffices that $R(C)$ does not overwhelm $\tau\cdot D(C)$.

\begin{proposition}[Confounding sign preservation --- proportional case]
\label{prop:conf_prop}
If $f(A) = \lambda\,g(A)$ for some constant $\lambda$ (i.e., the confounder effects on $T$ and $Y$ are proportional), then
\begin{equation}
  B(C) = \bigl(1-\xi(C)\bigr)\,B_{\mathrm{OV}},
  \label{eq:convex}
\end{equation}
where
\[
  \xi(C) = \frac{V_B\;\sigma_T^2}
  {V\bigl(V-V_B+\sigma_T^2\bigr)}
  \;\in\;[0,1],
\]
where $V_B=\operatorname{Var}(\mathbb{E}[g(A)\mid A^*])$ and $V=\operatorname{Var}(g(A))$.
Consequently, $B(C)$ lies between $0$ and $B_{\mathrm{OV}}$, and $\operatorname{sign}(\operatorname{plim}\hat\tau)=\operatorname{sign}(\tau)$ whenever $\operatorname{sign}(\tau)=\operatorname{sign}(\tau+B_{\mathrm{OV}})$.
\end{proposition}

\begin{proof}
When $f=\lambda g$: the between-group covariance in~\eqref{eq:lotc} becomes $\lambda\,\operatorname{Var}(\mathbb{E}[g\mid A^*])$, and $\operatorname{Cov}(g,f)=\lambda\,\operatorname{Var}(g)$.
So $R(C)=\lambda\bigl(\operatorname{Var}(g)-\operatorname{Var}(\mathbb{E}[g\mid A^*])\bigr)=\lambda\,V_W$, where $V_W=\operatorname{Var}(g)-V_B$ with $V_B=\operatorname{Var}(\mathbb{E}[g\mid A^*])$.
Similarly, $D(C)=V_W+\sigma_T^2$.
Hence
\[
  B(C) = \frac{\lambda\,V_W}{V_W+\sigma_T^2},\qquad
  B_{\mathrm{OV}} = \frac{\lambda\,V}{V+\sigma_T^2},
\]
where $V=\operatorname{Var}(g)=V_B+V_W$.
The ratio
\[
  \frac{B(C)}{B_{\mathrm{OV}}} = \frac{V_W(V+\sigma_T^2)}{V(V_W+\sigma_T^2)} = 1 - \xi(C)
\]
is a decreasing function of $V_B\in[0,V]$.
At $V_B=0$ (no classification, $C$ uniform): $\xi=0$, $B=B_{\mathrm{OV}}$.
At $V_B=V$ (perfect classification, $C=I$): $\xi=1$, $B=0$.
Since $0\leq V_B\leq V$, we have $\xi\in[0,1]$ and $B(C)\in[0,B_{\mathrm{OV}}]$ (with the sign of $B_{\mathrm{OV}}$).
\end{proof}

\begin{proposition}[Confounding sign preservation --- general case]
\label{prop:conf_gen}
For general $g$ and $f$ (not necessarily proportional), define the worst-case bias bound
\begin{equation}
  \bar{B} \;=\; \frac{\sqrt{\operatorname{Var}(g(A))\,\operatorname{Var}(f(A))}}{\sigma_T^2}.
  \label{eq:bbar}
\end{equation}
If $|\tau| > \bar{B}$, then $\operatorname{sign}(\operatorname{plim}\hat\tau(C))=\operatorname{sign}(\tau)$ for all column-stochastic $C$.
\end{proposition}

\begin{proof}
We need $\operatorname{sign}(\tau\cdot D(C)+R(C))=\operatorname{sign}(\tau)$, which holds if $|R(C)|<|\tau|\cdot D(C)$.
Since $D(C)\geq\sigma_T^2$ (shown in Lemma~\ref{lem:bias_decomp}), it suffices to show $|R(C)|<|\tau|\,\sigma_T^2$.

\emph{Step 1: Bound $|R(C)|$.}
$R(C)=\mathbb{E}[\operatorname{Cov}(g(A),f(A)\mid A^*)]$.
By the Cauchy--Schwarz inequality applied to each within-group covariance:
\begin{multline*}
  |\operatorname{Cov}(g,f\mid A^*{=}j)|\\
  \leq \sqrt{\operatorname{Var}(g\mid A^*{=}j)\,\operatorname{Var}(f\mid A^*{=}j)}.
\end{multline*}
Taking expectations and applying Cauchy--Schwarz again:
\begin{align*}
  |R(C)|
  &\leq \mathbb{E}\bigl[\sqrt{\operatorname{Var}(g\mid A^*)\,\operatorname{Var}(f\mid A^*)}\bigr]\\
  &\leq \sqrt{\mathbb{E}[\operatorname{Var}(g\mid A^*)]\;\mathbb{E}[\operatorname{Var}(f\mid A^*)]}\\
  &\leq \sqrt{\operatorname{Var}(g(A))\;\operatorname{Var}(f(A))},
\end{align*}
where the last step uses $\mathbb{E}[\operatorname{Var}(X\mid Y)]\leq\operatorname{Var}(X)$ (law of total variance).

\emph{Step 2: Conclude.}
The bias satisfies
\[
  |B(C)| = \frac{|R(C)|}{D(C)} \leq \frac{\sqrt{\operatorname{Var}(g)\,\operatorname{Var}(f)}}{\sigma_T^2} = \bar{B}.
\]
When $|\tau|>\bar{B}$, the bias cannot cancel $\tau$, so $\operatorname{sign}(\operatorname{plim}\hat\tau)=\operatorname{sign}(\tau)$ for all $C$.
\end{proof}

\paragraph{Relationship to the omitted-variable condition.}
A weaker (necessary) condition for sign preservation is $\operatorname{sign}(\tau)=\operatorname{sign}(\tau_{\mathrm{OV}})$ where $\tau_{\mathrm{OV}}=\tau+\operatorname{Cov}(g,f)/\operatorname{Var}(T)$.
If this fails, then $C=\frac{1}{K}\mathbf{1}\mathbf{1}^\top$ (uniform noise) already flips the sign.
The rigorous sufficient condition $|\tau|>\bar{B}$ implies the weaker condition, since $|B_{\mathrm{OV}}|=|\operatorname{Cov}(g,f)|/\operatorname{Var}(T)\leq\bar{B}$ by Cauchy--Schwarz and $\operatorname{Var}(T)\geq\sigma_T^2$.

\paragraph{Verification for paper parameters ($K{=}3$).}
With $\tau{=}2.0$, $\boldsymbol{\alpha}{=}(1.5,{-}1.0)$, $\boldsymbol{\beta}^{(A)}{=}(1.0,{-}0.5)$, $\sigma_T{=}1.0$, $\mathbf{p}{=}(0.40,0.35,0.25)$:
\begin{align*}
  \operatorname{Var}(g) &= \textstyle\sum_k p_k\alpha_k^2 - (\sum_k p_k\alpha_k)^2\\
  &= 0.962,\\
  \operatorname{Var}(f) &= \textstyle\sum_k p_k(\beta_k^{(A)})^2 - (\sum_k p_k\beta_k^{(A)})^2\\
  &= 0.362,\\
  \bar{B} &= \sqrt{0.962\times 0.362}/1.0^2 = 0.590.
\end{align*}
Since $|\tau|{=}2.0 > \bar{B}{=}0.590$, sign preservation is guaranteed for all column-stochastic $C$ by Proposition~\ref{prop:conf_gen}.
Moreover, $B_{\mathrm{OV}}=\operatorname{Cov}(g,f)/\operatorname{Var}(T)=0.588/1.962={+}0.300$, giving $\tau_{\mathrm{OV}}{=}2.300>0$, confirming that both the necessary and sufficient conditions are satisfied.

\paragraph{Extension to $K > 3$.}
For $K{>}3$, treatment effect coefficients are drawn from a fixed pool with alternating signs and decreasing magnitudes:
\begin{multline*}
  \boldsymbol{\beta} = (0,\; 1.0,\; {-}0.5,\; 0.7,\; {-}0.3,\\
  0.5,\; {-}0.2,\; 0.4,\; {-}0.15,\; 0.3,\; \ldots),
\end{multline*}
truncated to $K$ entries. The category distribution uses $p_k \propto 1/(k{+}1)$, normalized to sum to~1. Other topology-specific parameters (confounding strengths, mediator effects, etc.)\ follow analogous alternating-sign pools.

\subsection{Proof of Theorem~\ref{thm:monotone} (Confounding Monotonicity)}
\label{app:monotonicity_proof}

Under the interpolation $C(\delta)=(1{-}\delta)I+\delta C_0$, we prove $|B(C(\delta))|$ is non-decreasing in $\delta$ for proportional effects ($f=\lambda g$) and provide numerical verification for the general case.
The proof proceeds by direct algebraic analysis of $V_B(\delta)=\operatorname{Var}(\mathbb{E}[g(A)\mid A^*_\delta])$ as a rational function of $\delta$, establishing convexity and bounding the derivative at both endpoints.

\paragraph{Proportional case ($f=\lambda g$).}
From Proposition~\ref{prop:conf_prop}, $B(C)=\lambda V_W/(V_W+\sigma_T^2)$ where $V_W=\operatorname{Var}(g)-V_B$ and $V_B(\delta)=\operatorname{Var}(\mathbb{E}[g(A)\mid A^*_\delta])$.

\emph{Step 1: Convexity of $V_B(\delta)$.}
Define $N_j(\delta)=\sum_k g_k\,C(\delta)_{jk}\,p_k$ and $q_j(\delta)=\sum_k C(\delta)_{jk}\,p_k$.
Both are affine in $\delta$, with $q_j(\delta)>0$.
Then $V_B(\delta)=\sum_j N_j(\delta)^2/q_j(\delta) - \bar{g}^2$.
Each term $N_j^2/q_j$ is the composition of the perspective function $(x,t)\mapsto x^2/t$ (convex for $t>0$) with the affine map $\delta\mapsto(N_j(\delta),\,q_j(\delta))$, hence convex in $\delta$.
A sum of convex functions is convex, so $V_B(\delta)$ is convex on $[0,\delta_{\max}]$.

\emph{Step 2: Derivative formula.}
Let $\mu_j(\delta)=N_j(\delta)/q_j(\delta)=\mathbb{E}[g(A)\mid A^*{=}j]$.
Then
\begin{equation}
\frac{dV_B}{d\delta}
= -\sum_{j,k}\bigl[(C_0)_{jk}-\mathbf{1}_{j=k}\bigr]\,p_k\,(g_k-\mu_j(\delta))^2.
\label{eq:dVB}
\end{equation}
\emph{Derivation.}
$\frac{d}{d\delta}\frac{N_j^2}{q_j}=2\mu_j\dot{N}_j-\mu_j^2\dot{q}_j$
where $\dot{N}_j=m_j^0-g_jp_j$ and $\dot{q}_j=q_j^0-p_j$ (with $m_j^0=\sum_k g_k(C_0)_{jk}p_k$, $q_j^0=\sum_k(C_0)_{jk}p_k$).
Substituting and using column-stochasticity ($\sum_j(C_0)_{jk}=\sum_j\mathbf{1}_{j=k}=1$) to cancel the $g_k^2$ terms yields~\eqref{eq:dVB}.

\emph{Step 3: $V_B'(0)\leq 0$ (universal, no conditions on $C_0$).}
At $\delta{=}0$: $\mu_j(0)=g_j$ (since $C(0)=I$), so $(g_k-\mu_j(0))^2=(g_k-g_j)^2$.
The diagonal terms ($j{=}k$) vanish.
Hence
\[
V_B'(0)=-\sum_{j\neq k}(C_0)_{jk}\,p_k\,(g_j-g_k)^2\;\leq\; 0,
\]
since every term is non-negative.

\emph{Step 4: $V_B'(1)\leq 0$ for $K{=}2$.}
For $K{=}2$ with $C_0=\bigl[\begin{smallmatrix}a&1{-}b\\1{-}a&b\end{smallmatrix}\bigr]$ and $a,b\geq\tfrac{1}{2}$,
set $g_0{=}0$, $g_1{=}1$ without loss of generality
(the general case scales by $(\Delta g)^2$).
At $\delta{=}1$: $q_0{=}ap_0{+}(1{-}b)p_1$, $q_1{=}(1{-}a)p_0{+}bp_1$,
$\mu_0{=}(1{-}b)p_1/q_0$, $\mu_1{=}bp_1/q_1$.
Expanding~\eqref{eq:dVB} at $\delta{=}1$ and factoring:
\begin{align*}
  V_B'(1) &= (1{-}a)\,p_0(\mu_0^2{-}\mu_1^2)\\
  &\quad{}+(1{-}b)\,p_1\bigl[(1{-}\mu_1)^2{-}(1{-}\mu_0)^2\bigr] \\
  &= (\mu_0{-}\mu_1)\,\bigl[(1{-}a)\,p_0(\mu_0{+}\mu_1)\\
  &\quad{}+(1{-}b)\,p_1(2{-}\mu_0{-}\mu_1)\bigr].
\end{align*}
The bracket is non-negative (since $0\leq\mu_j\leq 1$ and all coefficients are non-negative).
Direct calculation gives $\mu_0{-}\mu_1=p_0\,p_1(1{-}a{-}b)/(q_0\,q_1)$
(expand $(1{-}b)\,q_1-b\,q_0=p_0(1{-}a{-}b)$).
Under $a,b\geq\tfrac{1}{2}$: $a{+}b\geq 1$, so $\mu_0{-}\mu_1\leq 0$ and $V_B'(1)\leq 0$.

\emph{Step 5: Monotonicity from convexity.}
Since $V_B$ is convex (Step~1), its derivative $V_B'$ is non-decreasing.
When $V_B'(1)\leq 0$, we have $V_B'(\delta)\leq V_B'(1)\leq 0$ for all $\delta\in[0,1]$, so $V_B$ is non-increasing.

\emph{Step 6: $|B(\delta)|$ is non-decreasing.}
$V_W(\delta)=\operatorname{Var}(g)-V_B(\delta)$ is non-decreasing (Step~5).
The function $h(x)=|\lambda|\,x/(x+\sigma_T^2)$ is strictly increasing for $x\geq 0$.
Hence $|B(\delta)|=h(V_W(\delta))$ is non-decreasing. \qed

\paragraph{Condition on $C_0$.}
The proof requires $V_B'(1)\leq 0$, which is guaranteed for $K{=}2$ when $(C_0)_{jj}\geq\tfrac{1}{2}$ (Step~4).
For $K\geq 3$, $V_B'(1)\leq 0$ holds for all $g$ and $p$ if and only if a certain matrix $M-W$ (depending on $C_0$ and $p$) is negative semi-definite; simple diagonal thresholds are sufficient in practice but not universal.
Numerical verification across 20{,}000 random $(C_0,p,g)$ triples per $K\in\{3,5,7\}$ with $(C_0)_{jj}\geq 1/K$ finds worst-case $|B|$ step deviations $< 0.013$ (Table~\ref{tab:conjecture_verify}), confirming $\varepsilon$-approximate monotonicity.
A $K{=}2$ counterexample ($C_0$ the permutation matrix, $(C_0)_{jj}{=}0$) shows the diagonal condition is necessary: $V_B(\delta)=(\delta{-}\tfrac{1}{2})^2$ increases on $(\tfrac{1}{2},1]$.

\paragraph{General case (non-proportional effects).}
$|B(C(\delta))|\leq\bar{B}(\delta)$ where $\bar{B}(\delta)=\sqrt{\mathbb{E}[\operatorname{Var}(g\mid A^*_\delta)]\,\mathbb{E}[\operatorname{Var}(f\mid A^*_\delta)]}\,/\,D(\delta)$ (Proposition~\ref{prop:conf_gen}).
By the same convexity argument, each $\mathbb{E}[\operatorname{Var}(\cdot\mid A^*_\delta)]$ is non-decreasing in $\delta$, and $D(\delta)=\mathbb{E}[\operatorname{Var}(T\mid A^*_\delta)]$ is non-decreasing.
Extended verification (20{,}000 random $C_0$ matrices, $K \in \{3,5,7\}$) reveals worst-case step deviations of 0.011 (confounding), confirming approximate tightness; a strict monotonicity proof for the general case remains open.

\begin{remark}[Conditional exact monotonicity]
\label{rem:conditional_monotonicity}
Exact monotonicity ($V_B'(\delta) \leq 0$ for all $\delta$) holds under the \emph{component-wise diagonal dominance} condition:
\begin{equation}
  \alpha_j \geq \frac{q_j}{p_j + q_j} \quad \text{for all } j,
  \label{eq:dd_condition}
\end{equation}
where $\alpha_j = (C_0)_{jj}$ and $q_j = \sum_k (C_0)_{jk} p_k$. This condition requires that each class's classification accuracy exceeds a threshold determined by its spillover ratio $q_j/p_j$. The proof uses Cauchy--Schwarz on the endpoint derivative decomposition $V_B'(1) = \sum_j (p_j + q_j)\,e_j^2 - \sum_{j \neq k} (C_0)_{jk}\,p_k\,(g_j - g_k)^2$, where $e_j = g_j - \mu_j$ (full proof in supplementary material).

For doubly stochastic $C_0$ with uniform $p$, condition~\eqref{eq:dd_condition} reduces to $\alpha_j \geq 1/2$, which for $K{=}2$ recovers the exact condition in Theorem~\ref{thm:monotone}. Numerical testing ($10^5$ random column-stochastic matrices with $\alpha_j \geq 0.5$) shows the condition holds in $99.3\%$ of cases; the $0.7\%$ violations arise predominantly under class imbalance with maximum step deviation $< 0.001$. When the condition fails, a quantitative bound $V_B'(1) \leq (1{-}c)^2 R_g^2$ (where $c = \min_j \alpha_j$ and $R_g = \max g_k - \min g_k$) ensures violations vanish as classifier accuracy improves.
\end{remark}

\subsection{Mediation Topology ($A$ = Mediator)}
\label{app:mediation}

\paragraph{Data-generating process.}
$T\to A\to Y$, $T\to Y$.
\begin{align*}
  T &\sim \operatorname{Bernoulli}(0.5),\\
  A\mid T &\sim \operatorname{Categorical}\bigl(\operatorname{softmax}(\boldsymbol{\gamma}+\boldsymbol{\delta}\,T)\bigr),\\
  Y &= \tau_{\mathrm{dir}}\,T + f(A) + \varepsilon_Y.
\end{align*}

\paragraph{Bias decomposition.}
By Lemma~\ref{lem:bias_decomp}, the direct-effect estimator satisfies:
\begin{equation}
  \operatorname{plim}\hat\tau_{\mathrm{dir}}(C) = \tau_{\mathrm{dir}} + \frac{\mathbb{E}[\operatorname{Cov}(T,\,f(A)\mid A^*)]}{D(C)}.
  \label{eq:med_bias}
\end{equation}

The key structural difference from confounding: $T$ causes $A$, so the residual covariance $\operatorname{Cov}(T,f(A)\mid A^*)$ captures the indirect effect of $T$ on $Y$ through $A$ that is not absorbed by conditioning on $A^*$.

\paragraph{Endpoint analysis.}

At $C=I$ ($A^*{=}A$): conditioning on $A$ blocks the mediated path $T\to A\to Y$, so $\operatorname{Cov}(T,f(A)\mid A)=0$ and $\operatorname{plim}\hat\tau=\tau_{\mathrm{dir}}$.

At $C=\frac{1}{K}\mathbf{1}\mathbf{1}^\top$ ($A^*\perp A$): no information about $A$ is used, and
\begin{align*}
  \mathbb{E}[\operatorname{Cov}(T,f(A)\mid A^*)] &= \operatorname{Cov}(T,f(A)),\\
  D(C) &= \operatorname{Var}(T) = 0.25.
\end{align*}
The indirect effect is $\tau_{\mathrm{ind}}=\mathbb{E}[f(A)\mid T{=}1]-\mathbb{E}[f(A)\mid T{=}0]$, and
\[
  \operatorname{Cov}(T,f(A)) = 0.25\,\tau_{\mathrm{ind}}.
\]
Hence at $C$ uniform: $\operatorname{plim}\hat\tau = \tau_{\mathrm{dir}}+\tau_{\mathrm{ind}} = \tau_{\mathrm{total}}$.

\paragraph{Numerical Verification (Mediation sign preservation).}
\label{find:med_sign}
$\operatorname{sign}(\operatorname{plim}\hat\tau_{\mathrm{dir}})=\operatorname{sign}(\tau_{\mathrm{dir}})$ whenever $\operatorname{sign}(\tau_{\mathrm{dir}})=\operatorname{sign}(\tau_{\mathrm{total}})$, where $\tau_{\mathrm{total}}=\tau_{\mathrm{dir}}+\tau_{\mathrm{ind}}$.
This result is verified numerically for $100{,}000$ independently sampled column-stochastic matrices $C$; a closed-form proof for general $K$ remains open.

\begin{proof}
Let $R(C)=\mathbb{E}[\operatorname{Cov}(T,f(A)\mid A^*)]$ and $D(C)=\mathbb{E}[\operatorname{Var}(T\mid A^*)]>0$.

\emph{Endpoints.}
At $C{=}I$: $f(A)$ is constant given $A{=}A^*$, so $R(I)=0$ and $B(I)=0$.
At $C=\frac{1}{K}\mathbf{1}\mathbf{1}^\top$: $R=\operatorname{Cov}(T,f(A))=0.25\,\tau_{\mathrm{ind}}$ and $D=\operatorname{Var}(T)=0.25$, giving $B=\tau_{\mathrm{ind}}$.

\emph{Within-group structure for binary $T$.}
Since $T\in\{0,1\}$, the within-$A^*$-group bias admits a revealing decomposition.
Let $p_j=P(T{=}1\mid A^*{=}j)$.
Then $\operatorname{Var}(T\mid A^*{=}j)=p_j(1{-}p_j)$ and
\begin{multline*}
  \operatorname{Cov}(T,f(A)\mid A^*{=}j)\\
  = p_j(1{-}p_j)\bigl[\mathbb{E}[f(A)\mid T{=}1,A^*{=}j]\\
  \quad{}-\mathbb{E}[f(A)\mid T{=}0,A^*{=}j]\bigr].
\end{multline*}
Define $\delta_j=\mathbb{E}[f(A)\mid T{=}1,A^*{=}j]-\mathbb{E}[f(A)\mid T{=}0,A^*{=}j]$ as the \emph{conditional indirect effect} in the $A^*{=}j$ subpopulation.
Then
\begin{align*}
  B(C)&=\frac{R(C)}{D(C)}
  =\frac{\sum_j q_j\,p_j(1{-}p_j)\,\delta_j}{\sum_j q_j\,p_j(1{-}p_j)}\\
  &=\textstyle\sum_j w_j\,\delta_j,
\end{align*}
a weighted average of the conditional indirect effects, with weights
$w_j=q_j p_j(1{-}p_j)\big/\!\sum_\ell q_\ell p_\ell(1{-}p_\ell)\geq 0$
summing to~1.

At $C{=}I$: each $A^*{=}j$ group has $A{=}j$ a.s., so $\delta_j=f(j)-f(j)=0$ and $B=0$.
At $C$ uniform: $\delta_j=\tau_{\mathrm{ind}}$ for all $j$ and $B=\tau_{\mathrm{ind}}$.

\emph{Sign preservation.}
When $\operatorname{Cov}(g(A),f(A))>0$ (the indirect effect reinforces the direct effect, i.e., $\tau_{\mathrm{dir}}$ and $\tau_{\mathrm{ind}}$ have the same sign), the bias $B(C)\geq 0$ cannot oppose $\tau_{\mathrm{dir}}$, so sign is preserved trivially.

When $\tau_{\mathrm{dir}}$ and $\tau_{\mathrm{ind}}$ have opposite signs but $|\tau_{\mathrm{ind}}|<|\tau_{\mathrm{dir}}|$ (i.e., $\operatorname{sign}(\tau_{\mathrm{dir}})=\operatorname{sign}(\tau_{\mathrm{total}})$), sign reversal requires some $|\delta_j|>|\tau_{\mathrm{dir}}|$ with the opposing sign.
Since each $\delta_j$ is a difference of weighted averages of $f$ values (with weights $P(A{=}k\mid T{=}t,A^*{=}j)$), and $B(C)$ is a convex combination of $\delta_j$, the bound $|B(C)|\leq\max_j|\delta_j|$ holds.
We verify numerically that $\max_j|\delta_j|\leq|\tau_{\mathrm{ind}}|$ for all 100{,}000 random $C$ matrices in our experiments, confirming sign preservation.
\end{proof}

\paragraph{Verification for paper parameters ($K{=}3$).}
With $\tau_{\mathrm{dir}}{=}0.5$, $\boldsymbol{\gamma}{=}(0,0,0)$, $\boldsymbol{\delta}{=}(0,0.8,{-}0.6)$, $\boldsymbol{\beta}{=}(0,1.0,{-}0.5)$:

$\tau_{\mathrm{ind}}=\mathbb{E}[f(A)\mid T{=}1]-\mathbb{E}[f(A)\mid T{=}0]=0.351$ (computed from the softmax class probabilities).
$\tau_{\mathrm{total}}=0.5+0.351=0.851>0$.
Since $\tau_{\mathrm{dir}}{>}0$ and $\tau_{\mathrm{total}}{>}0$, sign preservation holds for all $C$.

\paragraph{Mediation monotonicity (proportional effects).}
\label{par:med_monotone}

Under the interpolation $C(\delta)=(1{-}\delta)I+\delta C_0$, we prove $|B(C(\delta))|$ is non-decreasing in $\delta$ when the mediator's effect on $Y$ is proportional to its informativeness about $T$: $f(A)=\lambda\,g(A)$ for some scalar $\lambda$, where $g(k)=\mathbb{E}[T\mid A{=}k]$.

\emph{Step 1: Decompose $T$ via the mediator.}
Define $g(A)=\mathbb{E}[T\mid A]$ and $\varepsilon_T = T - g(A)$.
By construction, $\mathbb{E}[\varepsilon_T\mid A]=0$.
By non-differential misclassification ($T\perp A^*\mid A$):
$\mathbb{E}[\varepsilon_T\mid A{=}k, A^*{=}j] = 0$,
so $\mathbb{E}[\varepsilon_T\mid A^*{=}j]=0$ and $\operatorname{Cov}(\varepsilon_T,\,g(A)\mid A^*{=}j)=0$ for every $j$.

\emph{Step 2: Reduce the numerator.}
Since $f=\lambda g$:
\begin{align*}
  R(C) &= \lambda\,\mathbb{E}[\operatorname{Var}(g(A)\mid A^*)] = \lambda\,V_W.
\end{align*}

\emph{Step 3: Constant additive term in denominator.}
\begin{align*}
  D(C) &= \mathbb{E}[\operatorname{Var}(T\mid A^*)]
       = V_W + \sigma_\varepsilon^2,
\end{align*}
where $\sigma_\varepsilon^2\equiv\mathbb{E}[\operatorname{Var}(\varepsilon_T\mid A^*)]$.
This is constant (independent of $C$):
$\mathbb{E}[\operatorname{Var}(\varepsilon_T\mid A^*)]
  = \sum_k p_k\,\operatorname{Var}(T\mid A{=}k)
  = \mathbb{E}[\operatorname{Var}(T\mid A)]$,
where the equality uses column-stochasticity: $\sum_j q_j\,P(A{=}k\mid A^*{=}j)=p_k$.

\emph{Step 4: Closed form and monotonicity.}
$B(C) = \lambda\,V_W/(V_W+\sigma_\varepsilon^2)$,
structurally identical to confounding (Proposition~\ref{prop:conf_prop}) with $\sigma_\varepsilon^2$ playing the role of $\sigma_T^2$.
Since $V_B(\delta)$ is convex with $V_B'(0)\leq 0$ (same argument as \S\ref{app:monotonicity_proof}, Steps~1--3), $V_W(\delta)=V{-}V_B(\delta)$ is non-decreasing under the same conditions on $C_0$.
Since $h(x)=|\lambda|x/(x+\sigma_\varepsilon^2)$ is increasing, $|B(C(\delta))|$ is non-decreasing.

\paragraph{Boundedness of the leakage coefficient.}

The mediation bias~\eqref{eq:mediation} writes the residual indirect effect as $(1-\lambda(C))\,\tau_{\mathrm{ind}}$, which (for $\tau_{\mathrm{ind}}\neq 0$) implicitly defines
\begin{align}
\label{eq:lambda_def}
  \lambda(C) &\;:=\; 1-\frac{B(C)}{\tau_{\mathrm{ind}}},\\
  B(C)&\;\equiv\;\operatorname{plim}\hat{\tau}_{\mathrm{dir}}(A^*)-\tau_{\mathrm{dir}},\notag\\
  \tau_{\mathrm{ind}}&\;=\;\frac{\operatorname{Cov}(T,f(A))}{\operatorname{Var}(T)}.\notag
\end{align}
Substituting back into~\eqref{eq:mediation} gives the convex-combination form
$\operatorname{plim}\hat\tau_{\mathrm{dir}}(C)=\lambda(C)\,\tau_{\mathrm{dir}}+(1-\lambda(C))\,\tau_{\mathrm{total}}$, so $\lambda(C)$ is the weight placing the biased plim between $\tau_{\mathrm{dir}}$ (perfect classification) and $\tau_{\mathrm{total}}$ (no information).

\begin{proposition}[Mediation leakage boundedness]
\label{prop:lambda_bound}
Let $\lambda(C)$ be defined by~\eqref{eq:lambda_def} with $\tau_{\mathrm{ind}}\neq 0$, and let $C$ be column-stochastic.
\begin{enumerate}
\item[(a)] $\lambda(C)\leq 1$ iff $\operatorname{sign}(B(C))=\operatorname{sign}(\tau_{\mathrm{ind}})$ (the bias points in the direction of the indirect effect; no overcorrection).
\item[(b)] $\lambda(C)\geq 0$ iff additionally $|B(C)|\leq|\tau_{\mathrm{ind}}|$ (the bias does not exceed the indirect effect in magnitude).
\end{enumerate}
Equivalently, $\lambda(C)\in[0,1]$ iff $\operatorname{plim}\hat\tau_{\mathrm{dir}}(C)\in[\min(\tau_{\mathrm{dir}},\tau_{\mathrm{total}}),\,\max(\tau_{\mathrm{dir}},\tau_{\mathrm{total}})]$.

Under \emph{proportional effects} ($f(A)=\lambda_0\,g(A)$ with $g(k)=\mathbb{E}[T\mid A{=}k]$), both (a) and (b) hold for every $C$, so $\lambda(C)\in[0,1]$ unconditionally.
For \emph{general (non-proportional)} $f$, both (a) and (b) can fail because $\operatorname{Cov}(\mathbb{E}[g\mid A^*],\mathbb{E}[f\mid A^*])$ may differ in sign or magnitude from $\operatorname{Cov}(g,f)$ (a covariance-level analogue of Simpson's paradox under $A^*$ aggregation); the numerical verification below quantifies the empirical rate.
The endpoints $\lambda(I)=1$ and $\lambda(C)\to 0$ in the rank-one (uninformative) limit hold in all regimes.

\emph{Sign preservation.} When $\lambda(C)\in[0,1]$, the convex-combination form gives $\operatorname{plim}\hat\tau_{\mathrm{dir}}(C)\in[\min(\tau_{\mathrm{dir}},\tau_{\mathrm{total}}),\,\max(\tau_{\mathrm{dir}},\tau_{\mathrm{total}})]$. If additionally $\operatorname{sign}(\tau_{\mathrm{dir}})=\operatorname{sign}(\tau_{\mathrm{total}})$, both endpoints share the same sign, so $\operatorname{sign}(\operatorname{plim}\hat\tau_{\mathrm{dir}})=\operatorname{sign}(\tau_{\mathrm{dir}})$ for every~$C$. Under proportional effects, $\lambda(C)\in[0,1]$ is automatic (above), so sign preservation of $\hat\tau_{\mathrm{dir}}$ reduces to the single condition $\operatorname{sign}(\tau_{\mathrm{dir}})=\operatorname{sign}(\tau_{\mathrm{total}})$.
\end{proposition}

\begin{proof}
\emph{(a),(b).} With $\tau_{\mathrm{ind}}\neq 0$, $\lambda(C)\leq 1\iff B(C)/\tau_{\mathrm{ind}}\geq 0\iff \operatorname{sign}(B(C))=\operatorname{sign}(\tau_{\mathrm{ind}})$. Given (a), $\lambda(C)\geq 0\iff B(C)/\tau_{\mathrm{ind}}\leq 1\iff|B(C)|\leq|\tau_{\mathrm{ind}}|$.

\emph{Proportional case.} Step~4 of the mediation derivation gives $B(C)=\lambda_0\,V_W(C)/(V_W(C)+\sigma_\varepsilon^2)$; the rank-one limit $V_W{\to}V$ yields $\tau_{\mathrm{ind}}=\lambda_0\,V/(V+\sigma_\varepsilon^2)$. Both quantities carry $\operatorname{sign}(\lambda_0)$, establishing~(a). The ratio
\[
  \frac{B(C)}{\tau_{\mathrm{ind}}}\;=\;\frac{V_W(C)\,(V+\sigma_\varepsilon^2)}{V\,(V_W(C)+\sigma_\varepsilon^2)}
\]
is monotone increasing in $V_W(C)\in[0,V]$ with range $[0,1]$ (its derivative in $V_W$ is $\sigma_\varepsilon^2(V+\sigma_\varepsilon^2)/[V(V_W+\sigma_\varepsilon^2)^2]>0$), establishing~(b). Hence $\lambda(C)\in[0,1]$ unconditionally.

\emph{General case.} Substituting $\lambda(C)$ from~\eqref{eq:lambda_def} into~\eqref{eq:mediation} yields $\operatorname{plim}\hat\tau_{\mathrm{dir}}(C)=\lambda(C)\,\tau_{\mathrm{dir}}+(1{-}\lambda(C))\,\tau_{\mathrm{total}}$, so $\lambda(C)\in[0,1]$ iff $\operatorname{plim}\hat\tau_{\mathrm{dir}}(C)$ lies in the closed interval with endpoints $\tau_{\mathrm{dir}}$ and $\tau_{\mathrm{total}}$.

\emph{Endpoints.} At $C{=}I$, $A^*{=}A$ a.s., $\mathbb{E}[T\mid A^*]=g(A)$, and FWL gives $\operatorname{plim}\hat\tau_{\mathrm{dir}}=\tau_{\mathrm{dir}}$, hence $B(I)=0$ and $\lambda(I)=1$. At any rank-one $C$ (rows identical), $A^*\perp A$ so $\mathbb{E}[T\mid A^*]$ is constant, the partial regression on $A^*$ has no effect, and $\operatorname{plim}\hat\tau_{\mathrm{dir}}=\operatorname{Cov}(Y,T)/\operatorname{Var}(T)=\tau_{\mathrm{total}}$, giving $B=\tau_{\mathrm{ind}}$ and $\lambda=0$.
\end{proof}

\paragraph{Relation to the between-variance ratio.}
A natural \emph{diagnostic} closely related to $\lambda(C)$ is the between-variance ratio
\begin{equation}
\label{eq:vb_ratio}
  \lambda_V(C) \;:=\; \frac{V_B(C)}{V},
\end{equation}
where $V_B(C)=\operatorname{Var}\!\bigl(\mathbb{E}[g(A)\mid A^*]\bigr)$ and $V=\operatorname{Var}(g(A))$,
which satisfies $\lambda_V(C)\in[0,1]$ unconditionally by the law of total variance applied to $g(A)$. Under proportional effects, $\lambda$ and $\lambda_V$ agree at the endpoints and are related by
\begin{multline*}
  1-\lambda(C)\;=\;\frac{V_W(C)\,(V+\sigma_\varepsilon^2)}{V\,(V_W(C)+\sigma_\varepsilon^2)}\\
  =\;\bigl(1-\lambda_V(C)\bigr)\cdot\frac{V+\sigma_\varepsilon^2}{V_W(C)+\sigma_\varepsilon^2},
\end{multline*}
so they coincide in the limit $\sigma_\varepsilon^2\gg V$. They generally differ for non-proportional $f$. We use $\lambda$ throughout the main text because it parameterizes~\eqref{eq:mediation} directly; $\lambda_V$ remains a useful summary of the channel quality of $A^*$ as a proxy for $A$ (independent of the structural form of $f$).

\paragraph{Numerical verification.}
For each $K\in\{3,5,7\}$ we draw $N{=}10{,}000$ configurations $(\pi,g,\sigma_\varepsilon^2,C)$ with $\pi\sim\operatorname{Dirichlet}(\mathbf{1})$, $g\sim\mathcal{N}(0,I_K)$, $\sigma_\varepsilon^2\sim\operatorname{Uniform}(0.1,2)$, columns of $C\sim\operatorname{Dirichlet}(\mathbf{1})$, and evaluate $\lambda(C)$ from~\eqref{eq:lambda_def} in two regimes for $f$:
\begin{itemize}\itemsep0pt
\item \textbf{Proportional} ($f=\lambda_0\,g$, $\lambda_0\sim\mathcal{N}(0,1)\!\setminus\!\{0\}$): $\lambda(C)\in[0,1]$ in all $30{,}000$ trials (pooled observed range $[1.2{\times}10^{-5},\,0.850]$; zero out-of-bounds), confirming the closed-form bound. Sign preservation of $\hat\tau_{\mathrm{dir}}$ therefore reduces to $\operatorname{sign}(\tau_{\mathrm{dir}})=\operatorname{sign}(\tau_{\mathrm{total}})$.
\item \textbf{Non-proportional} ($f\sim\mathcal{N}(0,I_K)$ drawn independently of $g$, restricted to $|\tau_{\mathrm{ind}}|\geq 0.05$ to avoid the ill-conditioned regime where $\lambda{=}1{-}B/\tau_{\mathrm{ind}}$ diverges): $\lambda(C)$ exceeds $[0,1]$ in $\sim15$--$17\%$ of trials, illustrating that (a) and (b) are not automatic outside the proportional case.
\end{itemize}
The diagnostic $\lambda_V(C)$ lies in $[0,1]$ across all $30{,}000$ trials irrespective of regime, as guaranteed by the law of total variance.
Reproduction code and numerical results are included in the supplementary material.

\subsection{Collider Topology ($A$ = Collider)}
\label{app:collider}

\paragraph{Data-generating process.}
$T\to A\leftarrow Y$, $T\to Y$.
The treatment effect is unconfounded:
\begin{align*}
  T &\sim \mathcal{N}(0,1),\quad
  Y = \tau\,T + \varepsilon_Y,\\
  &\quad \varepsilon_Y\sim\mathcal{N}(0,1),\;\varepsilon_Y\perp T,\notag\\
  A\mid T,Y &\sim \operatorname{Categorical}\bigl(\operatorname{softmax}(\\
  &\qquad\boldsymbol{\delta}_T\,T + \boldsymbol{\delta}_Y\,Y)\bigr).
\end{align*}
The collider $A$ is a common effect of $T$ and $Y$.
The no-control estimator (regressing $Y$ on $T$ alone) is consistent: $\operatorname{plim}\hat\tau_{\mathrm{nc}}=\tau$.
The researcher erroneously controls for $A^*$, inducing Berkson's bias.

\paragraph{Bias decomposition.}
Unlike confounding, $\varepsilon_Y$ is not independent of $A$ (since $A$ depends on $Y=\tau T+\varepsilon_Y$), so Lemma~\ref{lem:bias_decomp} does not apply directly.
Instead, we derive the bias from the FWL identity~\eqref{eq:fwl}.
Writing $\tilde{Y}=\tau\tilde{T}+(\varepsilon_Y-\mathbb{E}[\varepsilon_Y\mid A^*])$ and noting that $\mathbb{E}[\tilde{T}\,\varepsilon_Y]=\mathbb{E}[\operatorname{Cov}(T,\varepsilon_Y\mid A^*)]$ (since $T\perp\varepsilon_Y$ unconditionally):
\begin{align}
  \operatorname{plim}\hat\tau(C) &= \tau + B(C),\notag\\
  B(C) &= \frac{\mathbb{E}[\operatorname{Cov}(T,\,\varepsilon_Y\mid A^*)]}{\mathbb{E}[\operatorname{Var}(T\mid A^*)]}.
  \label{eq:bias_col}
\end{align}
Under non-differential misclassification ($A^*\perp(T,\varepsilon_Y)\mid A$), define $g(k)=\mathbb{E}[T\mid A{=}k]$ and $h(k)=\mathbb{E}[\varepsilon_Y\mid A{=}k]$.
By the tower property, $\mathbb{E}[T\mid A^*]=\mathbb{E}[g(A)\mid A^*]$ and $\mathbb{E}[\varepsilon_Y\mid A^*]=\mathbb{E}[h(A)\mid A^*]$.
Since $\operatorname{Cov}(T,\varepsilon_Y)=0$, the law of total covariance gives:
\[
  B(C) = \frac{-\operatorname{Cov}\bigl(\mathbb{E}[g(A)\mid A^*],\;\mathbb{E}[h(A)\mid A^*]\bigr)}{\mathbb{E}[\operatorname{Var}(T\mid A^*)]}.
\]

\paragraph{Endpoint analysis.}
At $C=I$ ($A^*{=}A$):
$B(I) = -\operatorname{Cov}(g(A),h(A))\,/\,\mathbb{E}[\operatorname{Var}(T\mid A)] = B_{\mathrm{Berkson}}$,
the full Berkson's bias from conditioning on the collider.

At $C=\frac{1}{K}\mathbf{1}\mathbf{1}^\top$ ($A^*\perp A$):
$\mathbb{E}[g(A)\mid A^*]$ and $\mathbb{E}[h(A)\mid A^*]$ are constant, so $B=0$ and $\operatorname{plim}\hat\tau=\tau$.

This is the structural reverse of confounding: misclassification attenuates the collider-induced bias toward zero, rather than growing bias from zero toward $B_{\mathrm{OV}}$.

\begin{proposition}[Collider sign preservation --- proportional case]
\label{prop:col_prop}
If $h(A)=\lambda\,g(A)$ for some constant $\lambda$, then $|B(C)|\leq|B(I)|$ for all column-stochastic $C$, and $B(C)$ has the same sign as $B(I)$.
Consequently, $\operatorname{sign}(\operatorname{plim}\hat\tau)=\operatorname{sign}(\tau)$ whenever $\operatorname{sign}(\tau)=\operatorname{sign}(\tau+B_{\mathrm{Berkson}})$.
\end{proposition}

\begin{proof}
When $h=\lambda g$, the numerator becomes $-\lambda\,\operatorname{Var}(\mathbb{E}[g(A)\mid A^*])$ and the denominator is $\operatorname{Var}(T)-\operatorname{Var}(\mathbb{E}[g(A)\mid A^*])$.
Let $V^*=\operatorname{Var}(\mathbb{E}[g(A)\mid A^*])\in[0,V_g]$ where $V_g=\operatorname{Var}(g(A))$.
Then
\[
  |B(C)| = |\lambda|\,\frac{V^*}{\operatorname{Var}(T)-V^*}.
\]
The function $x\mapsto x/(a-x)$ is increasing on $[0,a)$.
With $a=\operatorname{Var}(T)>V_g\geq V^*$:
$|B(C)|\leq|\lambda|\,V_g/(\operatorname{Var}(T)-V_g)=|B(I)|$.
At $V^*{=}0$ (uniform $C$): $B{=}0$.  At $V^*{=}V_g$ ($C{=}I$): $B{=}B(I)$.
\end{proof}

\begin{proposition}[Collider sign preservation --- general case]
\label{prop:col_gen}
Define
\begin{equation}
  \bar{B}_{\mathrm{col}} = \frac{\sqrt{\operatorname{Var}(g(A))\,\operatorname{Var}(h(A))}}{\mathbb{E}[\operatorname{Var}(T\mid A)]}.
  \label{eq:bbar_col}
\end{equation}
If $|\tau|>\bar{B}_{\mathrm{col}}$, then $\operatorname{sign}(\operatorname{plim}\hat\tau(C))=\operatorname{sign}(\tau)$ for all column-stochastic $C$.
\end{proposition}

\begin{proof}
\emph{Step~1: Bound the numerator.}
By Cauchy--Schwarz:
$|\operatorname{Cov}(\mathbb{E}[g\mid A^*],\mathbb{E}[h\mid A^*])|\leq\sqrt{\operatorname{Var}(\mathbb{E}[g\mid A^*])\,\operatorname{Var}(\mathbb{E}[h\mid A^*])}$.
Since conditional expectation reduces variance: $\operatorname{Var}(\mathbb{E}[g\mid A^*])\leq\operatorname{Var}(g(A))$ and likewise for $h$.

\emph{Step~2: Bound the denominator.}
$\mathbb{E}[\operatorname{Var}(T\mid A^*)]=\operatorname{Var}(T)-\operatorname{Var}(\mathbb{E}[T\mid A^*])\geq\operatorname{Var}(T)-\operatorname{Var}(g(A))=\mathbb{E}[\operatorname{Var}(T\mid A)]$.

\emph{Step~3: Conclude.}
$|B(C)|\leq\sqrt{\operatorname{Var}(g)\,\operatorname{Var}(h)}\,/\,\mathbb{E}[\operatorname{Var}(T\mid A)]=\bar{B}_{\mathrm{col}}$.
When $|\tau|>\bar{B}_{\mathrm{col}}$, the bias cannot overwhelm $\tau$. In practice, $\operatorname{Var}(g(A))$ and $\operatorname{Var}(h(A))$ can be estimated from observable data by bootstrapping regression coefficients from $A^*$-stratified subsamples, yielding conservative upper bounds on $\bar{B}_{\mathrm{col}}$.
\end{proof}

\paragraph{Verification for paper parameters ($K{=}3$).}
With $\tau{=}1.0$, $\boldsymbol{\delta}_T{=}(0,\,0.5,\,{-}0.3)$, $\boldsymbol{\delta}_Y{=}(0,\,0.8,\,0.6)$, from $5{\times}10^6$-sample Monte Carlo:
\begin{align*}
  \operatorname{Var}(g(A)) &= 0.187,\quad
  \operatorname{Var}(h(A)) = 0.076,\\
  \mathbb{E}[\operatorname{Var}(T\mid A)] &= 0.813,\\
  \bar{B}_{\mathrm{col}} &= \sqrt{0.187\times 0.076}\;/\;0.813\\
  &= 0.147.
\end{align*}
Since $|\tau|{=}1.0\gg\bar{B}_{\mathrm{col}}{=}0.147$, Proposition~\ref{prop:col_gen} guarantees sign preservation.
The Berkson bias $B_{\mathrm{Berkson}}{=}{-}0.114$, giving $\tau+B_{\mathrm{Berkson}}{=}0.886{>}0$.
Verified across $100{,}000$ random column-stochastic $C$: $\tau_{\mathrm{sign\,flip}}{=}0\%$, and $|B(C)|\leq|B_{\mathrm{Berkson}}|$ in all cases.

\subsection{Front-door Topology ($A$ = Mediator under Confounding)}
\label{app:frontdoor}

\paragraph{Data-generating process.}
$T\to A\to Y$, $U\to T$, $U\to Y$.
An unobserved confounder $U$ affects both treatment and outcome:
\begin{align*}
  U &\sim \mathcal{N}(0,1),\quad
  T = \alpha_U\,U + \eta,\\
  &\quad \eta\sim\mathcal{N}(0,1),\;\eta\perp U,\notag\\
  A\mid T &\sim \operatorname{Categorical}\bigl(\operatorname{softmax}(\boldsymbol{\gamma}+\boldsymbol{\delta}\,T)\bigr),\\
  Y &= f(A) + \lambda\,U + \varepsilon_Y,\quad \varepsilon_Y\perp(T,A,U).
\end{align*}
There is no direct $T{\to}Y$ edge; the causal effect of $T$ on $Y$ operates entirely through $A$.
The researcher controls for $A^*$ in $Y\sim 1+T+D_1^*+\cdots+D_{K-1}^*$.

\paragraph{Bias decomposition.}
Since the direct effect is zero, the FWL coefficient on $T$ captures only bias:
\[
  \operatorname{plim}\hat\tau(C) = \frac{\mathbb{E}[\operatorname{Cov}(T,\,Y\mid A^*)]}{\mathbb{E}[\operatorname{Var}(T\mid A^*)]}.
\]
Substituting $Y=f(A)+\lambda U+\varepsilon_Y$ and using $\varepsilon_Y\perp(T,A,A^*)$:
\begin{align}
  \operatorname{plim}\hat\tau(C)
  &= \underbrace{\frac{\lambda\alpha_U}{\operatorname{Var}(T)}}_{\text{conf.}}
  + \underbrace{\frac{\mathbb{E}[\operatorname{Cov}(T,\!f(A)\mid A^*)]}{D(C)}}_{B_{\mathrm{med}}(C)}\!,
  \label{eq:fd_decomp}
\end{align}
where $D(C)=\mathbb{E}[\operatorname{Var}(T\mid A^*)]$.

\begin{proof}[Derivation of~\eqref{eq:fd_decomp}]
Expand the numerator as
$\mathbb{E}[\operatorname{Cov}(T,f(A)\mid A^*)] + \lambda\,\mathbb{E}[\operatorname{Cov}(T,U\mid A^*)]$.
For the confounding term: since $T=\alpha_U U+\eta$ with $(U,\eta)$ jointly Gaussian, the regression residual $\xi=U-\frac{\alpha_U}{\operatorname{Var}(T)}T$ satisfies $\xi\perp T$.
Because $A$ is a function of $T$ (plus independent softmax noise) and $A^*$ is a function of $A$ (plus independent misclassification noise), $\xi\perp(T,A,A^*)$.
Therefore
\[
  \operatorname{Cov}(T,U\mid A^*\!{=}j)
  = \tfrac{\alpha_U}{\operatorname{Var}(T)}\operatorname{Var}(T\mid A^*\!{=}j)
  \;\;\forall\,j,
\]
and $\lambda\,\mathbb{E}[\operatorname{Cov}(T,U\mid A^*)]=\frac{\lambda\alpha_U}{\operatorname{Var}(T)}\,D(C)$, yielding the constant first term after division by $D(C)$.
\end{proof}

This decomposition reveals that misclassification does not affect the confounding component.
The confusion matrix $C$ modulates only the mediation-type leakage $B_{\mathrm{med}}(C)$, which has the same structure as in~§\ref{app:mediation}.

\paragraph{Endpoint analysis.}
At $C=I$: $B_{\mathrm{med}}(I)=0$ (controlling $A$ blocks the $T\to A\to Y$ path), and $\operatorname{plim}\hat\tau(I)=\lambda\alpha_U/\operatorname{Var}(T)$ (pure confounding residual).

At $C=\frac{1}{K}\mathbf{1}\mathbf{1}^\top$: $B_{\mathrm{med}}=\operatorname{Cov}(T,f(A))/\operatorname{Var}(T)$, and $\operatorname{plim}\hat\tau_{\mathrm{nc}}=[\operatorname{Cov}(T,f(A))+\lambda\alpha_U]/\operatorname{Var}(T)$, the naive regression coefficient with no controls.

\paragraph{Numerical Verification (Front-door sign preservation).}
\label{find:fd_sign}
Since $B_{\mathrm{med}}(C)$ has the same structure as the mediation bias (§\ref{app:mediation}), interpolating between $0$ at $C{=}I$ and $\operatorname{Cov}(T,f(A))/\operatorname{Var}(T)$ at $C$ uniform. Sign preservation holds whenever the two endpoints have the same sign:
\[
  \operatorname{sign}\!\bigl(\lambda\alpha_U\bigr)
  = \operatorname{sign}\!\bigl(\lambda\alpha_U + \operatorname{Cov}(T,f(A))\bigr).
\]
This result is verified numerically for $100{,}000$ independently sampled column-stochastic matrices $C$.

\paragraph{Verification for paper parameters ($K{=}3$).}
With $\alpha_U{=}1.0$, $\lambda{=}1.0$, $\boldsymbol{\beta}{=}(0,1.0,{-}0.5)$, $\boldsymbol{\gamma}{=}(0,0.5,{-}0.3)$, $\boldsymbol{\delta}{=}(0,1.0,{-}0.8)$:
\begin{align*}
  \lambda\alpha_U/\operatorname{Var}(T) &= 1.0\!\times\!1.0/2.0 = 0.500,\\
  \operatorname{Cov}(T,\!f(A))/\operatorname{Var}(T) &= 0.594/2.0 = 0.297.
\end{align*}
Both components are positive, so $\operatorname{plim}\hat\tau(C)\in[0.500,\,0.797]>0$ for all $C$.

\subsection{Exposure Topology: Sign Reversal Counterexample}
\label{app:exposure}

\begin{proposition}[Exposure sign reversal]
\label{prop:exp_counter}
For $K{=}3$ with $A$ as the treatment variable, there exist a marginal distribution $\mathbf{p}$, treatment effects $\boldsymbol{\beta}$, and a column-stochastic $C$ with $C_{kk}>0.5$ for all $k$, such that $\operatorname{sign}(\operatorname{plim}\hat\beta_1)\neq\operatorname{sign}(\beta_1)$.
\end{proposition}

\begin{proof}[Construction]
Consider the exposure data-generating process (DGP) $Y = \beta_1 D_1 + \beta_2 D_2 + \varepsilon_Y$ with:
\begin{itemize}
  \item Uniform prior: $\mathbf{p}=(1/3,\,1/3,\,1/3)$.
  \item True effects: $\beta_1=1$,\; $\beta_2=-10$.
\end{itemize}
Choose the confusion matrix
\[
  C = \begin{pmatrix}
    0.60 & 0.10 & 0.10\\
    0.10 & 0.60 & 0.39\\
    0.30 & 0.30 & 0.51
  \end{pmatrix},
\]
with $C_{kk}\in\{0.60,\,0.60,\,0.51\}$, all ${>}\,0.5$.
Column sums are $1$ (verifiable by inspection).

The OLS estimator regresses $Y$ on $(1,D_1^*,D_2^*)$.
The plim of $\hat\beta_j$ equals $\mathbb{E}[Y\mid A^*{=}j]-\mathbb{E}[Y\mid A^*{=}0]$, where
\begin{align*}
  \mathbb{E}[Y\mid A^*{=}j] &= \sum_{k=0}^{2} P(A{=}k\mid A^*{=}j)\,\beta_k\\
  &= \sum_{k=1}^{2}\frac{C_{jk}\,p_k}{q_j}\,\beta_k.
\end{align*}

\emph{Marginal distribution of $A^*$:}
\begin{align*}
  q_0 &= \tfrac{1}{3}(0.60+0.10+0.10)=0.267,\\
  q_1 &= \tfrac{1}{3}(0.10+0.60+0.39)=0.363,\\
  q_2 &= \tfrac{1}{3}(0.30+0.30+0.51)=0.370.
\end{align*}

\emph{Conditional means:}
\begin{align*}
  \mathbb{E}[Y\mid A^*{=}0]
  &= \tfrac{0.10\cdot\frac{1}{3}\cdot 1 \;+\; 0.10\cdot\frac{1}{3}\cdot({-}10)}{0.267}\\
  &= \tfrac{0.0333-0.3333}{0.267} = -1.125,\\[4pt]
  \mathbb{E}[Y\mid A^*{=}1]
  &= \tfrac{0.60\cdot\frac{1}{3}\cdot 1 \;+\; 0.39\cdot\frac{1}{3}\cdot({-}10)}{0.363}\\
  &= \tfrac{0.200-1.300}{0.363} = -3.028,\\[4pt]
  \mathbb{E}[Y\mid A^*{=}2]
  &= \tfrac{0.30\cdot\frac{1}{3}\cdot 1 \;+\; 0.51\cdot\frac{1}{3}\cdot({-}10)}{0.370}\\
  &= \tfrac{0.100-1.700}{0.370} = -4.324.
\end{align*}

\emph{Estimated coefficient:}
\begin{align*}
  \operatorname{plim}\hat\beta_1
  &= \mathbb{E}[Y\mid A^*{=}1]-\mathbb{E}[Y\mid A^*{=}0]\\
  &= -3.028-(-1.125) = \mathbf{-1.903}.
\end{align*}
The true $\beta_1=+1>0$ but $\operatorname{plim}\hat\beta_1=-1.903<0$: \textbf{sign reversal}.

\emph{Mechanism.}
The large off-diagonal entry $C_{12}=0.39$ (probability of classifying true category~2 as $A^*{=}1$) routes substantial probability mass from category~2 ($\beta_2=-10$) into the $A^*{=}1$ bin.
Despite $C_{11}=0.60>0.5$, the contamination from the strongly negative $\beta_2$ overwhelms the positive $\beta_1$, reversing the sign of the estimated effect for category~1.
This failure mode is structurally impossible at $K{=}2$: a $2{\times}2$ column-stochastic matrix with $C_{kk}>0.5$ has the form $C=\bigl(\begin{smallmatrix}a&1{-}b\\1{-}a&b\end{smallmatrix}\bigr)$ with $a,b>0.5$, and the single coefficient $\hat\beta_1$ is a positive-weighted average of $\beta_1$, preserving sign.
For $K{\geq}3$, the additional degrees of freedom in $C$ permit cross-contamination between categories with opposite-sign effects.
\end{proof}

\paragraph{Structural argument for M-bias topology.}
\emph{M-bias:} The M-structure creates a backdoor path through two latent common causes.
Misclassification of $A$ weakens the statistical association along this backdoor path, reducing the spurious correlation rather than amplifying it.
The bias therefore interpolates between the full M-bias (perfect classification, $\delta{=}0$) and zero (no classification information, $\delta{\to}\delta_{\max}$), monotonically non-increasing in $\delta$.

\paragraph{M-bias sign-flip ASV $= \infty$.}
For the four covariate topologies (confounding, mediation, collider, M-bias), FWL invariance (Appendix~\ref{app:fwl}) implies that the coefficient on $T$ depends on $A^*$ only through the residualized relationship $\tilde{T} = T - \mathbb{E}[T \mid A^*]$. For M-bias, the spurious path through $A$ is the sole source of bias, and misclassification attenuates this path for all column-stochastic $C$. Since $\text{bias}(C(\delta))$ is monotonically non-increasing in $\delta$ and starts at a value with the same sign as $\tau$, no $\delta \in [0, \delta_{\max}]$ can produce a sign reversal. Therefore $\delta^*_{\text{sign}} = \infty$, which is a direct consequence of FWL invariance combined with the protective monotonicity of M-bias misclassification.

\paragraph{Remark.}
The sign preservation conditions for confounding, mediation, collider, and front-door are conditions on the DGP parameters (the relative strength of the causal effect versus the confounding, indirect, or collider-induced bias), not on the confusion matrix $C$.
Once the condition is satisfied, sign preservation holds for all column-stochastic $C$ with $C_{kk}>0$; no diagonal-dominance restriction is needed.
Conversely, if the condition fails (i.e., the omitted-variable or total-effect estimate has the opposite sign from $\tau$), then there exist confusion matrices that produce sign reversal.
For the DGP parameters used throughout our experiments, the conditions are comfortably satisfied.
The sign-alignment condition is violated when the confounder has opposing effects on treatment and outcome (e.g., $\gamma_A > 0$ on $T$ but $\gamma_A < 0$ on $Y$), producing $\operatorname{sign}(\tau) \neq \operatorname{sign}(\tau_{\mathrm{OV}})$. In applied settings, such opposing effects are uncommon when the analyst adjusts for $A$ precisely because they believe it confounds the $T$--$Y$ relationship in a consistent direction. When the condition fails, sign reversal becomes possible under sufficient misclassification, though this represents a pathological DGP where adjustment itself may be questionable.

\subsection{Proof of FWL Invariance}
\label{app:fwl_proof}

For the four topologies where $A$ enters as a covariate (confounding, mediation, collider, M-bias), the regression is $Y = \tau T + \boldsymbol{\gamma}^\top \mathbf{D}^* + \varepsilon$, where $\mathbf{D}^*$ is the $N \times (K{-}1)$ dummy matrix for $A^*$. Any matrix-based correction replaces $\mathbf{D}^*$ with $\tilde{\mathbf{D}} = \mathbf{D}^* M$ for some invertible $(K{-}1) \times (K{-}1)$ matrix $M$ (e.g., $M = C^{-1}$ for plug-in, or $M = (C^\top C)^{-1} C^\top$ for maximum likelihood estimation (MLE)). By the Frisch--Waugh--Lovell theorem~\cite{lovell1963}, $\hat{\tau}$ depends only on the residual of $T$ projected off $\operatorname{col}(\mathbf{D}^*)$. Since $\operatorname{col}(\mathbf{D}^* M) = \operatorname{col}(\mathbf{D}^*)$ for invertible $M$, the projection is unchanged and $\hat{\tau}(\tilde{\mathbf{D}}) = \hat{\tau}(\mathbf{D}^*)$ exactly. SIMEX adds noise $\mathbf{D}^*_{\text{sim}} = \mathbf{D}^* + \lambda \mathbf{E}$; when $\mathbf{E}$ is generated from the same column space, the extrapolation to $\lambda = -1$ converges to a column-space-equivalent transformation, yielding 0\% bias reduction empirically.

\paragraph{FWL breakdown under logistic regression.}
\label{sec:fwl_logistic}
The column-space argument applies only to estimators whose treatment coefficient is a function of $T$ residualized off $\operatorname{col}(\mathbf{D}^*)$, a property of linear projection. The logistic MLE solves $\sum_i (Y_i - \sigma(\mathbf{x}_i^\top\boldsymbol{\beta}))\mathbf{x}_i = \mathbf{0}$; the implicit weighting by $\sigma'(\cdot)$ depends on the actual values of the design columns, not only on their span, so replacing $\mathbf{D}^*$ by $\mathbf{D}^* M$ generally changes the fitted $\hat\tau$. We verify this on the confounding topology ($K{=}3$, $\tau{=}0.5$, $N{=}10{,}000\times 500$ MC reps) under both linear ($Y = \tau T + \mathbf{D}\boldsymbol{\beta}+\varepsilon$) and logistic ($Y\sim\mathrm{Bern}(\sigma(\tau T + \mathbf{D}\boldsymbol{\beta}))$) DGPs, comparing the uncorrected estimator (fit on $\mathbf{D}^*$) against Naive plug-in (fit on per-sample $(C^{-1}\mathbf{e}_{A^*})_{1:K}$). Per-rep estimates are paired (same $A^*$, same outcome).

$|\Delta|_{\max}=\max_m|\hat\tau_{\mathrm{corr}}^{(m)}-\hat\tau_{\mathrm{un}}^{(m)}|$ over 500 MC reps; bias reduction = $1-|\hat\tau_{\mathrm{corr}}-\tau_*|/|\hat\tau_{\mathrm{un}}-\tau_*|$. Confounding DAG, $K{=}3$, $N{=}10{,}000$.

\subsection{Non-Differential Misclassification Verification}
\label{sec:nondiff_verify}

The non-differential assumption $P(A^*{=}j \mid A{=}k, Y, \mathbf{Z}) = P(A^*{=}j \mid A{=}k)$ holds by design in standard LLM annotation pipelines: the model receives only source text as input and classifies without observing the outcome $Y$ or covariates $\mathbf{Z}$. The annotation API call is structurally independent of downstream variables, satisfying the conditional independence requirement.

To provide empirical support beyond the design argument, we examine topic-stratified confusion matrices using the VAST dataset ($K{=}3$, stance detection), where the \texttt{topic} field (e.g., ``gun control,'' ``college'') serves as a proxy for latent factors correlated with $Y$. Across 10 model configurations (3 LLMs $\times$ 2--4 prompt types) and 19 qualifying topics (minimum 10 samples per topic), we compute per-topic confusion matrices $C_{\text{topic}}(t) = P(A^*{=}j \mid A{=}k, \text{topic}{=}t)$ and compare against the overall $C = P(A^*{=}j \mid A{=}k)$.

Per-topic sample sizes range from 10 to 42, making individual topic CMs inherently noisy. A bootstrap permutation test ($B{=}10{,}000$) under the non-differential null shows that the expected grand maximum deviation $\max_{t,j,k} |C_{\text{topic}}(j,k) - C(j,k)|$ at the 95th percentile ranges from 0.90 to 0.98 across model configurations, driven by the smallest topics. For the 5 largest topics ($n \geq 28$), expected per-topic max deviation at the 95th percentile is 0.38--0.49, consistent with finite-sample variation rather than systematic topic-dependence. The observed per-topic variation is fully consistent with the non-differential null.

This proxy-based test is necessarily limited: topic is not the actual outcome $Y$, and small per-topic samples reduce statistical power. The design argument (LLM input excludes $Y$) remains the primary justification; the stress test in Section~\ref{sec:nondiff_stress} below quantifies bias degradation under hypothetical violations.

\input{exp_e_asv_stability_table.tex}

Three patterns emerge (Table~\ref{tab:asv_stability}). First, IV magnitude ASV is identical across all three DGPs ($\delta^*{=}0.203$). This is a mathematical identity, not an empirical finding: the Wald estimator $\hat{\tau}_{\text{IV}} = \text{RF}/\text{FS}$ has $\boldsymbol{\beta}$ in both numerator and denominator, and the ratio depends only on $C$ and the first-stage structure, making ASV invariant to $\boldsymbol{\beta}$ by construction. Confounding is consistently the most robust topology (highest ASV or no violation), and M-bias remains immune (not shown). However, the relative fragility of exposure vs.\ IV inverts with coefficient sign structure: under alternating signs, exposure is most fragile ($0.090 < 0.203$); under same-sign effects, IV is most fragile ($0.203 < 0.223$). Practitioners should compute ASV with their specific DGP rather than relying on generic rankings.

\subsection{Realistic Differential Misclassification Thresholds}
\label{sec:realistic_epsilon}

Our non-differential assumption (Definition~\ref{def:nondiff}) requires $P(A^*{\mid}A,Y) = P(A^*{\mid}A)$. We estimate how much $Y$-stratified differential error each of our 23 empirical LLM confusion matrices can tolerate before exposure sign reversal exceeds 5\%. For each matrix $C$, we construct $C_{\text{high}} = \text{renorm}(C + \varepsilon\Delta)$ and $C_{\text{low}} = \text{renorm}(C - \varepsilon\Delta)$ with random off-diagonal perturbation $\Delta$ (500 MC trials per $\varepsilon$), sweep $\varepsilon \in [0, 0.50]$, and report the safe threshold $\varepsilon^*_{\text{safe}} = \max\{\varepsilon : \text{reversal rate} < 5\%\}$ (Table~\ref{tab:realistic_epsilon}).

For $K{=}2$ (binary), no configuration produces sign reversal up to $\varepsilon{=}0.50$. For $K{\geq}3$, the majority of configurations are fragile ($\varepsilon^*_{\text{safe}}{=}0$), confirming that the non-differential assumption is load-bearing for multi-class exposure.

\subsection{Sign Reversal Determinants}
\label{sec:sign_determinants}

We investigate which confusion matrix features predict exposure sign reversal across the 23 empirical LLM configurations. For each matrix, we compute 10 structural features: $\min_k C_{kk}$ (minimum diagonal), $\overline{C_{kk}}$ (mean diagonal), $\max_{j \neq k} C_{jk}$ (maximum off-diagonal), $\max_{j \neq k} C_{jk}/C_{kk}$ (off-diagonal ratio), $\log(\kappa(C))$ (log condition number), mean and max column entropy, spectral gap, $\det(C)$, and minimum singular value. Sign reversal is defined as any exposure coefficient reversing sign under misclassification.

Best single-feature rule: $\det(C) < 0.014 \to$ sign reversal (accuracy 91.3\%, sensitivity 85.7\%, specificity 100\%). An AND rule ($\min_k C_{kk} < 0.55$ and mean entropy $> 0.40$) achieves 95.7\% accuracy with 100\% sensitivity.

\subsection{Y-Stratified Non-Differential Verification}
\label{sec:y_stratified_test}

We directly test the non-differential assumption (Definition~\ref{def:nondiff}) by stratifying confusion matrices by outcome-proxy variables and testing $C(Y{=}\text{high}) \approx C(Y{=}\text{low})$. Using 12 LLM configurations on VAST ($K{=}3$; 4 models $\times$ 3 prompting strategies, 1{,}500 samples each), we compute per-stratum confusion matrices under three Y-proxies and apply permutation tests ($B{=}10{,}000$, test statistic: $\max_{j,k} |C^{(0)}_{jk} - C^{(1)}_{jk}|$).

Content-independent Y (text length) shows no significant violation (1/12 rejection $\approx$ expected false positive rate at $\alpha{=}0.05$). Content-dependent proxies (topic, sentiment) show expected violations since LLMs observe text content.
